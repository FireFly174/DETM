def is_excluded_dir(path: Path, exclude_dirs: set[str]) -> bool:
    parts = {p.name for p in path.parents} | {path.name}
    return any(d in parts for d in exclude_dirs)


def matches_any(name: str, patterns: list[str]) -> bool:
    return any(fnmatch.fnmatch(name, p) for p in patterns)


def sha256_text(text: str) -> str:
    return hashlib.sha256(text.encode("utf-8", errors="ignore")).hexdigest()


def read_text_safely(p: Path) -> str:
    # Try UTF-8, then fallback.
    try:
        return p.read_text(encoding="utf-8")
    except UnicodeDecodeError:
        return p.read_text(encoding="utf-8", errors="replace")


def build_tree(root: Path, include: list[str], exclude_dirs: set[str]) -> str:
    lines: list[str] = []
    root = root.resolve()
    for dirpath, dirnames, filenames in os.walk(root):
        dpath = Path(dirpath)
        # prune excluded dirs
        dirnames[:] = [d for d in dirnames if d not in exclude_dirs]

        rel_dir = dpath.relative_to(root)
        indent = "  " * (len(rel_dir.parts) if rel_dir.parts != (".",) else 0)
        if str(rel_dir) != ".":
            lines.append(f"{indent}{rel_dir}/")

        for fn in sorted(filenames):
            if fn in DEFAULT_EXCLUDE_FILES:
                continue
            if matches_any(fn, include):
                lines.append(f"{indent}  {fn}")
    return "\n".join(lines)


def collect_files(root: Path, include: list[str], exclude_dirs: set[str]) -> list[Path]:
    root = root.resolve()
    out: list[Path] = []
    for dirpath, dirnames, filenames in os.walk(root):
        dpath = Path(dirpath)
        dirnames[:] = [d for d in dirnames if d not in exclude_dirs]

        for fn in filenames:
            if fn in DEFAULT_EXCLUDE_FILES:
                continue
            if matches_any(fn, include):
                p = dpath / fn
                out.append(p)
    return sorted(out, key=lambda p: str(p).lower())

    exclude_dirs = set(args.exclude_dir)
    include = list(args.include)

    files = collect_files(root, include, exclude_dirs)

    header.append(f"exclude_dirs: {sorted(exclude_dirs)}")
    header.append("")
    header.append("TREE:")
    header.append(build_tree(root, include, exclude_dirs))
    header.append("")
    header.append("=" * 80)
    header.append("FILES:")
    header.append("=" * 80)
    header.append("")

    chunks: list[str] = ["\n".join(header)]

    for p in files:
        rel = p.relative_to(root)
        size = p.stat().st_size
        if size > args.max_bytes:
            chunks.append(
                f"\n\n--- FILE: {rel} ---\n"
                f"(skipped: {size} bytes > max_bytes={args.max_bytes})\n"
            )
            continue

        text = read_text_safely(p)
        digest = sha256_text(text)[:16]

        chunks.append(

            + f"\nFILE: {rel}\nSIZE: {size} bytes\nSHA256_16: {digest}\n"
            + "-" * 80
            + "\n"
            + text
            + "\n"
        )

    out_path.write_text("".join(chunks), encoding="utf-8")
    print(f"[OK] Wrote bundle: {out_path}")
    print(f"[OK] Files included: {len(files)}")


if __name__ == "__main__":
    sys.exit(main())

# DETM — Discrete Entropy-Time Model

## Краткое описание

DETM — это исследовательская дискретная модель динамики на решётке, предназначенная для изучения возникновения устойчивых локализованных структур (инвариантов) из локальных правил переноса, без задания объектов, целей или глобальной координации.

Модель основана на сочетании:
- локального переноса скалярной величины (энергии),
- структурного (энтропийного) подавления,
- внутреннего времени (латентности).

Основной интерес представляет не поведение отдельных ячеек, а эмерджентные режимы и структуры, возникающие при коарсинге динамики.

---

## Что исследуется

В рамках проекта изучаются следующие вопросы:
- при каких условиях локальные правила приводят к устойчивым структурам;
- какие механизмы обеспечивают их долговечность и различимость;
- как внутренняя асинхронность влияет на устойчивость и масштабирование;
- существуют ли пределы плотности и пропускной способности таких структур.

---

## Что модель не утверждает

DETM не является физической, биологической или когнитивной теорией.
Модель не претендует на описание реальной Вселенной, сознания или фундаментальных законов природы.

Все интерпретации за пределами формальной динамики рассматриваются как внешние аналогии и не входят в канон проекта.

---

## Структура документации

DETM — базовая спецификация модели
1. Общий обзор

DETM (Discrete Entropy-Time Model, дискретная энтропийно-временная модель) — это дискретная динамическая система, заданная на регулярной решётке.
Модель описывает локальный перенос скалярной величины («энергии») с ограничениями, накладываемыми локальной структурой среды, энтропийным подавлением и внутренним временем (латентностью).

Цель модели — исследование возникновения устойчивых, локализованных и долгоживущих инвариантных структур, которые появляются исключительно из локальных правил взаимодействия, без задания объектов, целей или глобальной координации.

2. Пространство и время
2.1 Пространство

Система определена на двумерной решётке Ω размером N×N.
Каждая ячейка решётки идентифицируется позицией r, принадлежащей Ω.

Связи локальные: каждая ячейка взаимодействует только с ближайшими соседями (тип окрестности задаётся отдельно).

Граничные условия могут быть:

фиксированными,

открытыми,

периодическими (топология тора).

2.2 Время

Время дискретное и задаётся целыми шагами t = 0, 1, 2, …
Все обновления вычисляются относительно глобального тика.

При этом каждая ячейка может обладать собственным внутренним временем, отличающимся от глобального тика (см. раздел 5).

3. Переменные состояния

Каждая ячейка r в момент времени t имеет следующий набор состояний.

3.1 Энергия

E(r,t) ∈ [0, 1]

Энергия — скалярная величина, которая локально переносится между соседними ячейками.
Энергия является сохраняемой величиной (с учётом граничных условий).

3.2 Энтропийное подавление

S(r,t) ≥ 0

S — локальная скалярная величина, описывающая подавление переноса энергии.
Она зависит от:

отклонения энергии E(r,t) от опорного уровня E₀;

локальной неоднородности (разностей энергии с соседями).

S не является термодинамической энтропией.
Это эффективный структурный фактор, ограничивающий транспорт.

3.3 Внутреннее время (латентность)

τ(r,t) ≥ 0

τ описывает локальную задержку реакции ячейки.
Чем больше τ, тем медленнее через ячейку может проходить перенос энергии.

Внутреннее время не изменяет глобальный тик, а модулирует локальную динамику.

4. Потоки энергии
4.1 Базовый поток

Для каждой пары соседних ячеек (r, r′) определяется базовый поток энергии, пропорциональный разности энергий между ними.

Интенсивность потока масштабируется параметром проводимости κ > 0.

4.2 Эффективный поток

Реальный (эффективный) поток подавляется локальной структурой среды и внутренним временем.

Эффективный поток уменьшается:

при росте S(r,t);

при росте τ(r,t).

Таким образом:

перенос всегда локален;

перенос всегда ограничен;

в структурно сложных областях поток замедляется нелинейно.

5. Правило обновления энергии

Энергия в каждой ячейке обновляется по дискретному уравнению непрерывности:

Энергия в момент t+1 равна энергии в момент t минус сумма эффективных потоков, уходящих из ячейки к её соседям.

Это правило:

сохраняет суммарную энергию системы;

обеспечивает строгую локальность взаимодействий;

предотвращает неограниченное распространение возмущений.

6. Инвариантные структуры

Инвариантная структура определяется операционально, через наблюдаемые свойства динамики.

Структура считается инвариантной, если она удовлетворяет следующим условиям:

Устойчивость
Структура сохраняется на длительных временных интервалах.

Локализация
Структура занимает ограниченную область решётки.

Фазовая согласованность
Внутренняя динамика структуры характеризуется устойчивым периодом T.

Робастность
Структура сохраняет идентичность при умеренных внешних возмущениях.

Инварианты не задаются явно и не кодируются в правилах модели — они возникают эмерджентно.

7. Минимальность модели

Следующие элементы считаются необходимыми для возникновения инвариантов:

дискретная решётка;

скалярное энергетическое поле;

локальный перенос энергии с ограниченной пропускной способностью;

энтропийное подавление переноса;

внутренняя временная латентность.

Удаление любого из этих элементов разрушает наблюдаемое формирование устойчивых структур.

8. Область применимости

Данный документ описывает только базовые динамические правила модели DETM.

Любые интерпретации, аналогии, уровневые описания, сопоставления с физикой, биологией, когнитивными или инженерными системами рассматриваются отдельно и не входят в спецификацию модели.

# Каналы ускоренного согласования

## Локальные модификации среды

Канал определяется как локальная область, в которой параметры среды изменены так, что:
- внутреннее время снижено,
- энтропийное подавление уменьшено,
- пропускная способность увеличена.

Каналы не являются отдельными объектами, а представляют собой конфигурации параметров среды.

---

## Эффект каналов

Наличие канала приводит к уменьшению времени согласования между удалёнными областями, если динамика этих областей уже частично согласована.

Важно подчеркнуть, что канал не переносит энергию или информацию сам по себе, а лишь снижает латентность перестройки существующих режимов.

---

## Ограничения эффекта

Эффект каналов:
- требует заранее подготовленной среды,
- не нарушает локальности правил,
- исчезает при разрушении структурной поддержки.

Каналы не обеспечивают перенос в пустой или несогласованной среде.

# Коарсинг и уровни динамики

## Общая идея

Коарсинг в модели DETM — это процесс укрупнения согласованных областей динамики, возникающий эмерджентно в результате локальных правил переноса энергии, энтропийного подавления и внутреннего времени.

Коарсинг не вводится явно и не задаётся алгоритмически. Он наблюдается как следствие согласования фаз и периодов между соседними областями.

---

## Локальное согласование

На ранних стадиях динамики каждая ячейка эволюционирует в основном независимо, реагируя на локальные градиенты энергии.

По мере эволюции:
- локальные различия сглаживаются,
- области с близкими параметрами начинают вести себя согласованно,
- формируются домены с общим характерным периодом.

---

## Формирование уровней

Уровень определяется как область, в которой:
- внутренняя динамика синхронизирована по фазе,
- наблюдается единый характерный период,
- локальные флуктуации не разрушают согласованность.

Уровни не являются объектами модели, а используются как способ описания наблюдаемой структуры динамики.

---

## Глобальный тик и относительное время

Глобальный тик задаёт базовую дискретизацию времени, однако внутри уровня динамика может протекать с другим эффективным масштабом времени.

Переход к более крупному уровню можно интерпретировать как перенормировку времени, при которой период нижнего уровня становится единицей измерения на следующем уровне.

---

## Роль коарсинга

Коарсинг является необходимым механизмом для:
- формирования устойчивых инвариантов,
- появления объектоподобного поведения,
- ограничения плотности структур на уровне.

# Внутреннее время и асинхронность

## Понятие внутреннего времени

Внутреннее время τ — это локальная характеристика ячейки, определяющая задержку реакции на входящие потоки энергии.

Внутреннее время не изменяет глобальный тик, но модулирует скорость локальной перестройки состояния.

---

## Асинхронность динамики

Различия во внутреннем времени между ячейками приводят к асинхронности:
- разные области реагируют с разной скоростью,
- формируются фазовые сдвиги,
- возникает возможность локального согласования без глобальной синхронизации.

Асинхронность является ключевым фактором устойчивости сложных структур.

---

## Роль в устойчивости

Наличие внутреннего времени:
- предотвращает мгновенное распространение возмущений,
- подавляет резонансные разрушения,
- позволяет структурам адаптироваться к изменениям среды.

При отсутствии внутреннего времени динамика становится чрезмерно жёсткой и склонной к разрушению структур.

# Инвариантные структуры

## Определение

Инвариантная структура — это устойчивая локализованная конфигурация динамики, сохраняющая идентичность во времени и при умеренных возмущениях.

Инварианты не кодируются явно и не задаются начальными условиями. Они возникают как результат самоорганизации.

---

## Характерные свойства

Инвариантная структура обладает следующими признаками:
- пространственная локализация,
- устойчивый внутренний период,
- фазовая согласованность,
- робастность к локальным воздействиям.

Наличие всех признаков одновременно является необходимым условием для идентификации инварианта.

---

## Инварианты как объекты

Инварианты могут демонстрировать поведение, аналогичное объектам:
- сохранять форму при перемещении,
- реагировать на градиенты параметров,
- взаимодействовать с другими инвариантами.

При этом инварианты не являются частицами и не имеют массы или траектории в физическом смысле.

---

## Границы существования

Инварианты существуют только в определённых диапазонах параметров. При выходе за эти диапазоны структура:
- распадается,
- теряет периодичность,
- или растворяется в фоновом шуме.

Это позволяет экспериментально исследовать области устойчивости.

Критерий опровержения:

Инвариантные структуры характеризуются устойчивым внутренним периодом, согласованным по фазе внутри структуры.

устойчивые структуры не демонстрируют воспроизводимого периода или фазовой согласованности.

любое малое возмущение приводит к разрушению или радикальной перестройке структуры.

H6. Объектоподобное поведение инвариантов

Инварианты ведут себя как квазичастицы: они могут перемещаться, взаимодействовать с градиентами параметров и сохранять форму при движении.

структуры либо полностью неподвижны, либо разрушаются при любой попытке смещения.

При периодических граничных условиях инварианты могут демонстрировать устойчивое перемещение по решётке.

H8. Каналы пониженной латентности

# Метрики и наблюдаемые величины

## Общие принципы

Все эксперименты в модели DETM опираются на количественные метрики, извлекаемые из динамики системы.

Метрика считается корректной, если она:
- воспроизводима при повторных прогонах,
- устойчива к малым изменениям параметров,
- не зависит от визуальной интерпретации.

---

1. Энергия  
   Локальное значение энергии и её распределение по решётке.

   Интегральная величина, характеризующая суммарную интенсивность переносов энергии.

#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import argparse
import os
import numpy as np
import matplotlib.pyplot as plt


def div2d(Jx, Jy):
    # Jx, Jy: [N,N]
    # div = dJx/dx + dJy/dy (дискретно через центральную разность)
    dJx = 0.5 * (np.roll(Jx, -1, axis=0) - np.roll(Jx, 1, axis=0))
    dJy = 0.5 * (np.roll(Jy, -1, axis=1) - np.roll(Jy, 1, axis=1))
    return dJx + dJy


def curl2d(Jx, Jy):
    # curl_z = dJy/dx - dJx/dy
    dJy_dx = 0.5 * (np.roll(Jy, -1, axis=0) - np.roll(Jy, 1, axis=0))
    dJx_dy = 0.5 * (np.roll(Jx, -1, axis=1) - np.roll(Jx, 1, axis=1))
    return dJy_dx - dJx_dy


def save_heatmap(img, title, out_png):
    plt.figure(figsize=(7, 6))
    plt.imshow(img, origin="lower", interpolation="nearest")
    plt.title(title)
    plt.colorbar()
    plt.tight_layout()
    plt.savefig(out_png, dpi=160)
    plt.close()


def rings_indices(N, center, r_max=None):
    cx, cy = center
    xs = np.arange(N)[:, None]
    ys = np.arange(N)[None, :]
    rr = np.sqrt((xs - cx) ** 2 + (ys - cy) ** 2)
    if r_max is None:
        r_max = int(rr.max())
    ridx = np.clip(rr.astype(int), 0, r_max)
    return ridx, r_max


def ring_means(field, ridx, r_max):
    # field: [T,N,N] or [N,N]
    if field.ndim == 2:
        out = np.zeros((r_max + 1,), dtype=np.float64)
        for r in range(r_max + 1):
            m = (ridx == r)
            out[r] = field[m].mean() if m.any() else np.nan
        return out
    else:
        T = field.shape[0]
        out = np.zeros((T, r_max + 1), dtype=np.float64)
        for r in range(r_max + 1):
            m = (ridx == r)
            if not m.any():
                out[:, r] = np.nan
            else:
                out[:, r] = field[:, m].mean(axis=1)
        return out


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--npz", required=True, help="fields_hist_<run_id>.npz")

    args = ap.parse_args()

    data = np.load(args.npz)

    t = data.get("t", None)
    if t is None or (not np.isfinite(t).any()):
        # fallback
        any_key = next(k for k in data.files if k.endswith("_hist"))
        T = data[any_key].shape[0]
        t = np.arange(T, dtype=float)

    # load fields
    E_hist = data.get("E_hist", None)
    Jx_hist = data.get("Jx_hist", None)
    Jy_hist = data.get("Jy_hist", None)

    if E_hist is None:
        raise RuntimeError("E_hist not found in npz. Enable fields ('E') in logger.")

    # IMPORTANT: stats in float64
    E_hist = E_hist.astype(np.float32, copy=False)

    out_dir = args.out or os.path.join(os.path.dirname(args.npz), "analysis_fields")
    os.makedirs(out_dir, exist_ok=True)

    mean_E = np.mean(E_hist, axis=0, dtype=np.float64)

    # center

        idx = np.argmax(mean_E)

    ridx, r_max = rings_indices(N, (cx, cy), r_max=args.rmax)

    # ring signals for E
    E_ring = ring_means(E_hist, ridx, r_max)  # [T, r]
    save_heatmap(E_ring.T, "E ring means (r x t)", os.path.join(out_dir, "E_ring.png"))

    # J analysis if present
    if (Jx_hist is not None) and (Jy_hist is not None):
        Jx_hist = Jx_hist.astype(np.float32, copy=False)
        Jy_hist = Jy_hist.astype(np.float32, copy=False)

        # magnitude
        Jmag_hist = np.sqrt(Jx_hist * Jx_hist + Jy_hist * Jy_hist, dtype=np.float32)
        mean_J = np.mean(Jmag_hist, axis=0, dtype=np.float64)
        std_J = np.std(Jmag_hist, axis=0, dtype=np.float64)
        save_heatmap(mean_J, "mean(|J|)", os.path.join(out_dir, "mean_Jmag.png"))
        save_heatmap(std_J, "std(|J|)", os.path.join(out_dir, "std_Jmag.png"))

        # div/curl for last frame (or average)
        Jx_last = Jx_hist[-1]
        Jy_last = Jy_hist[-1]
        div_last = div2d(Jx_last, Jy_last)
        curl_last = curl2d(Jx_last, Jy_last)
        save_heatmap(div_last, "div J (last frame)", os.path.join(out_dir, "divJ_last.png"))
        save_heatmap(curl_last, "curl J (last frame)", os.path.join(out_dir, "curlJ_last.png"))

        # ring signals for |J|
        J_ring = ring_means(Jmag_hist, ridx, r_max)  # [T, r]
        save_heatmap(J_ring.T, "|J| ring means (r x t)", os.path.join(out_dir, "J_ring.png"))

    # plot a few ring curves
    plt.figure(figsize=(12, 5))
    for r in [0, 2, 5, 10, 15]:
        if r <= r_max:
            plt.plot(t, E_ring[:, r], label=f"r={r}")
    plt.title(f"E ring means around center ({cx},{cy})")
    plt.xlabel("t")
    plt.ylabel("E_ring")
    plt.legend()
    plt.tight_layout()
    plt.savefig(os.path.join(out_dir, "E_ring_curves.png"), dpi=160)
    plt.close()

    print("[OK] center:", (cx, cy), "r_max:", r_max)
    print("[OK] wrote:", out_dir)


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import argparse
import os
import numpy as np
import matplotlib.pyplot as plt


def dominant_freq_map(E_hist: np.ndarray, dt: float = 1.0) -> tuple[np.ndarray, np.ndarray]:
    """
    E_hist: [T, N, N]
    Возвращает:
      f_dom: [N, N] доминирующая частота (в 1/dt)
      p_dom: [N, N] мощность на доминирующей частоте
    """
    T = E_hist.shape[0]
    X = E_hist.astype(np.float32)

    # remove mean per cell to kill DC
    X = X - X.mean(axis=0, keepdims=True)

    # rFFT along time axis
    F = np.fft.rfft(X, axis=0)  # [F, N, N]
    P = (F.real * F.real + F.imag * F.imag)  # power

    freqs = np.fft.rfftfreq(T, d=dt)  # [F]

    # ignore DC bin (0)
    if P.shape[0] > 1:
        P0 = P[1:]
        idx = np.argmax(P0, axis=0) + 1
    else:
        idx = np.zeros((E_hist.shape[1], E_hist.shape[2]), dtype=int)

    f_dom = freqs[idx]
    p_dom = P[idx, np.arange(E_hist.shape[1])[:, None], np.arange(E_hist.shape[2])[None, :]]
    return f_dom, p_dom


def plot_top_points(t: np.ndarray, E_hist: np.ndarray, score_map: np.ndarray, k: int, out_png: str):
    """
    Рисует k точек с максимальным score_map (например std или p_dom).
    """
    N = score_map.shape[0]
    flat = score_map.reshape(-1)
    top_idx = np.argsort(flat)[-k:][::-1]

    plt.figure(figsize=(12, 6))
    for m, idx in enumerate(top_idx):
        i = idx // N
        j = idx % N
        y = E_hist[:, i, j]
        plt.plot(t, y, label=f"({i},{j}) score={score_map[i,j]:.4g}")

        if m >= 12:  # чтобы легенда не стала адом
            break

    plt.title(f"Top points by score (showing up to 13 of {k})")
    plt.xlabel("t (saved frames index / t_global if available)")
    plt.ylabel("E")
    plt.legend(fontsize=8)
    plt.tight_layout()
    plt.savefig(out_png, dpi=160)
    plt.close()


def save_heatmap(img: np.ndarray, title: str, out_png: str):

    # IMPORTANT: если сохранено в float16, статистики в numpy тоже будут float16 и могут быть мусором.
    # Апкастим для корректных mean/std/fft.
    E_hist = E_hist.astype(np.float32, copy=False)

    # базовые карты

#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import sys
from pathlib import Path

def read_patch(path: Path) -> list[dict]:
    blocks = []
    cur = None
    buf = []

    def flush():
        nonlocal cur, buf
        if cur:
            cur["content"] = "\n".join(buf).rstrip() + "\n"
            blocks.append(cur)
        cur, buf = None, []

    for line in path.read_text(encoding="utf-8").splitlines():
        if line.startswith("=== CREATE FILE ==="):
            flush()
            cur = {"type": "create"}
        elif line.startswith("=== MODIFY FILE ==="):
            flush()
            cur = {"type": "modify"}
        elif line.startswith("=== END PATCH ==="):
            flush()
            break
        elif line.strip() == "---":
            continue
        elif cur is not None and ":" in line and not buf:
            k, v = line.split(":", 1)
            cur[k.strip()] = v.strip()
        else:
            buf.append(line)

    flush()
    return blocks


def apply_patch(blocks: list[dict], root: Path):
    for b in blocks:
        path = root / b["path"]

        if b["type"] == "create":
            print(f"[CREATE] {path}")
            path.parent.mkdir(parents=True, exist_ok=True)
            if path.exists():
                print(f"  ! exists, skipped")
                continue
            path.write_text(b["content"], encoding="utf-8")

        elif b["type"] == "modify":
            if not path.exists():
                print(f"[ERROR] file not found: {path}")
                continue

            text = path.read_text(encoding="utf-8")

            mode = b.get("mode", "append")

            if mode == "append":
                anchor = b["anchor"]
                idx = text.find(anchor)
                if idx < 0:
                    print(f"[ERROR] anchor not found in {path}")
                    continue
                insert_at = idx + len(anchor)
                text = text[:insert_at] + "\n\n" + b["content"] + text[insert_at:]

            elif mode == "replace_block":
                a = b["anchor_start"]
                z = b["anchor_end"]
                i1 = text.find(a)
                i2 = text.find(z)
                if i1 < 0 or i2 < 0 or i2 <= i1:
                    print(f"[ERROR] block anchors not found in {path}")
                    continue
                i2 += len(z)
                text = text[:i1] + b["content"] + text[i2:]

            else:
                print(f"[ERROR] unknown mode {mode}")
                continue

            path.write_text(text, encoding="utf-8")
            print(f"[MODIFY] {path}")

def main():
    if len(sys.argv) != 2:
        print("Usage: python apply_patch.py patch.txt")
        sys.exit(1)

    patch_path = Path(sys.argv[1])
    root = Path("../../LMU").resolve()

    blocks = read_patch(patch_path)
    apply_patch(blocks, root)
    print("\n[OK] Patch applied")

if __name__ == "__main__":
    main()

class SummaryWriter:
    def __init__(self, path: str):
        self.path = path
        self._file = open(path, "a", newline="", encoding="utf-8")
        self._writer = None

    def write(self, row: dict):
        if self._writer is None:
            self._writer = csv.DictWriter(self._file, fieldnames=list(row.keys()))
            if self._file.tell() == 0:
                self._writer.writeheader()
        self._writer.writerow(row)
        self._file.flush()

    def close(self):
        self._file.close()

def bundle_series_timeseries(series_dir: str, out_name: str = "timeseries_bundle.csv") -> str:
    ts_dir = os.path.join(series_dir, "timeseries")
    paths = sorted(glob.glob(os.path.join(ts_dir, "timeseries_*.csv")))
    if not paths:
        raise FileNotFoundError(f"No timeseries found in {ts_dir}")

    frames = []
    for ordinal, p in enumerate(paths):
        df = pd.read_csv(p)

        run_id = os.path.basename(p)[len("timeseries_"):-4]
        if "run_id" not in df.columns:
            df.insert(0, "run_id", run_id)
        else:
            df["run_id"] = df["run_id"].fillna(run_id)

        if "t_rel" not in df.columns:
            t0 = float(df["t_global"].iloc[0])
            df["t_rel"] = df["t_global"] - t0

        df.insert(1, "run_ordinal", ordinal)
        frames.append(df)

    # harmonize columns
    cols = []

        for c in df.columns:
            if c not in cols:

        for c in cols:
            if c not in df.columns:
                df[c] = np.nan
        frames[i] = df[cols]

    out_path = os.path.join(series_dir, out_name)
    pd.concat(frames, ignore_index=True).to_csv(out_path, index=False)
    return out_path

def is_excluded_dir(path: Path, exclude_dirs: set[str]) -> bool:
    parts = {p.name for p in path.parents} | {path.name}
    return any(d in parts for d in exclude_dirs)


def matches_any(name: str, patterns: list[str]) -> bool:
    return any(fnmatch.fnmatch(name, p) for p in patterns)


def sha256_text(text: str) -> str:
    return hashlib.sha256(text.encode("utf-8", errors="ignore")).hexdigest()


def read_text_safely(p: Path) -> str:
    # Try UTF-8, then fallback.
    try:
        return p.read_text(encoding="utf-8")
    except UnicodeDecodeError:
        return p.read_text(encoding="utf-8", errors="replace")


def build_tree(root: Path, include: list[str], exclude_dirs: set[str]) -> str:
    lines: list[str] = []
    root = root.resolve()
    for dirpath, dirnames, filenames in os.walk(root):
        dpath = Path(dirpath)
        # prune excluded dirs
        dirnames[:] = [d for d in dirnames if d not in exclude_dirs]

        rel_dir = dpath.relative_to(root)
        indent = "  " * (len(rel_dir.parts) if rel_dir.parts != (".",) else 0)
        if str(rel_dir) != ".":
            lines.append(f"{indent}{rel_dir}/")

        for fn in sorted(filenames):
            if fn in DEFAULT_EXCLUDE_FILES:
                continue
            if matches_any(fn, include):
                lines.append(f"{indent}  {fn}")
    return "\n".join(lines)


def collect_files(root: Path, include: list[str], exclude_dirs: set[str]) -> list[Path]:
    root = root.resolve()
    out: list[Path] = []
    for dirpath, dirnames, filenames in os.walk(root):
        dpath = Path(dirpath)
        dirnames[:] = [d for d in dirnames if d not in exclude_dirs]

        for fn in filenames:
            if fn in DEFAULT_EXCLUDE_FILES:
                continue
            if matches_any(fn, include):
                p = dpath / fn
                out.append(p)
    return sorted(out, key=lambda p: str(p).lower())

    header.append(f"include: {include}")
    header.append(f"exclude_dirs: {sorted(exclude_dirs)}")
    header.append("")
    header.append("TREE:")

    header.append("=" * 80)

    out_path.write_text("".join(chunks), encoding="utf-8")
    print(f"[OK] Wrote bundle: {out_path}")

if __name__ == "__main__":
    sys.exit(main())

def _infer_run_id(path: str, df: pd.DataFrame) -> str:
    # 1) try column run_id
    if "run_id" in df.columns and df["run_id"].notna().any():
        v = df["run_id"].dropna().astype(str).iloc[0]
        if v:
            return v
    # 2) try filename timeseries_<runid>.csv
    base = os.path.basename(path)
    if base.startswith("timeseries_") and base.lower().endswith(".csv"):
        mid = base[len("timeseries_"):-len(".csv")]
        if mid:
            return mid
    # 3) fallback: basename without ext
    return os.path.splitext(base)[0]


def _estimate_avg_row_bytes(path: str, sample_rows: int = 300) -> int:
    # Read small sample and estimate bytes per row when written as CSV
    df = pd.read_csv(path, nrows=sample_rows)
    if df.empty:
        return 200  # arbitrary
    s = df.to_csv(index=False)
    # Rough: total bytes / rows (including header). Add small safety.
    bpr = max(50, int(len(s.encode("utf-8")) / max(1, len(df))) + 10)
    return bpr


def _count_rows_fast(path: str) -> int:
    # Fast line count without loading whole CSV
    # subtract header line if present
    with open(path, "rb") as f:
        n = sum(1 for _ in f)
    return max(0, n - 1)


def _pick_event_indices(df: pd.DataFrame) -> np.ndarray:
    """
    Событийные точки, если соответствующие колонки существуют.
    Возвращает массив индексов (позиции строк), которые важно сохранить.
    """
    idx = set()

    n = len(df)
    if n == 0:
        return np.array([], dtype=int)

    # Always keep first/last
    idx.add(0)
    idx.add(n - 1)

    # x_line change points
    if "x_line" in df.columns:
        x = df["x_line"].to_numpy()
        # treat -1 / NaN as missing
        x2 = np.copy(x)
        # mark changes
        ch = np.where(np.diff(x2, prepend=x2[0]) != 0)[0]
        for i in ch:
            idx.add(int(i))
            if i > 0:
                idx.add(int(i - 1))
            if i < n - 1:
                idx.add(int(i + 1))

    # Peaks of A (keep top K peaks by prominence-ish)
    if "A" in df.columns:
        A = df["A"].to_numpy(dtype=float)
        if np.isfinite(A).any():
            # local maxima: A[i-1] < A[i] >= A[i+1]
            m = np.zeros(n, dtype=bool)
            m[1:-1] = (A[1:-1] > A[:-2]) & (A[1:-1] >= A[2:])
            peak_idx = np.where(m)[0]
            if len(peak_idx) > 0:
                # rank by amplitude, keep up to 60
                K = min(60, len(peak_idx))
                top = peak_idx[np.argsort(A[peak_idx])[-K:]]
                for i in top:
                    idx.add(int(i))

    # Big jumps in T (period changes)
    if "T" in df.columns:
        T = df["T"].to_numpy(dtype=float)
        # ignore NaN
        if np.isfinite(T).any():
            Tn = np.copy(T)
            # forward-fill NaN for diff detection
            mask = np.isfinite(Tn)
            if mask.any():
                last = Tn[mask][0]
                for i in range(n):
                    if np.isfinite(Tn[i]):
                        last = Tn[i]
                    else:
                        Tn[i] = last
                d = np.abs(np.diff(Tn, prepend=Tn[0]))
                # keep top jump points
                K = min(40, n)
                top = np.argsort(d)[-K:]
                for i in top:
                    idx.add(int(i))

    # Max |dE_obj| events
    if "E_obj" in df.columns:
        Eo = df["E_obj"].to_numpy(dtype=float)
        if np.isfinite(Eo).any():
            d = np.abs(np.diff(Eo, prepend=Eo[0]))
            K = min(40, n)
            top = np.argsort(d)[-K:]
            for i in top:
                idx.add(int(i))

    return np.array(sorted(idx), dtype=int)


def _uniform_indices(n: int, k: int) -> np.ndarray:
    if n <= 0:
        return np.array([], dtype=int)
    if k >= n:
        return np.arange(n, dtype=int)
    # evenly spaced incl endpoints
    return np.unique(np.linspace(0, n - 1, k, dtype=int))


def compress_timeseries(
    df: pd.DataFrame,
    target_rows: int,
    min_uniform_rows: int = 80,
) -> pd.DataFrame:
    """
    Сжатие:
    - сохраняем событийные точки
    - добавляем равномерную сетку по времени
    - если всё ещё слишком много — уплотняем равномерную часть, но события оставляем
    """
    n = len(df)
    if n == 0:
        return df

    target_rows = max(10, int(target_rows))

    ev = _pick_event_indices(df)
    # Uniform backbone
    uni = _uniform_indices(n, max(min_uniform_rows, target_rows // 2))
    keep = np.unique(np.concatenate([ev, uni]))

    if len(keep) > target_rows:
        # Keep all events, and sample remaining uniformly from non-event indices
        ev_set = set(map(int, ev.tolist()))
        non_ev = np.array([i for i in keep if int(i) not in ev_set], dtype=int)

        # budget for non-events
        budget = max(0, target_rows - len(ev_set))
        if budget <= 0:
            keep2 = np.array(sorted(ev_set), dtype=int)
        else:
            sampled = _uniform_indices(len(non_ev), budget)
            keep2 = np.unique(np.concatenate([np.array(sorted(ev_set), dtype=int), non_ev[sampled]]))
        keep = keep2

    out = df.iloc[keep].copy()

    # Ensure time-ordered by t_rel if present, else index
    if "t_rel" in out.columns:
        out = out.sort_values("t_rel", kind="mergesort")
    else:
        out = out.sort_index(kind="mergesort")

    out.reset_index(drop=True, inplace=True)
    return out


def plan_all_runs(paths: List[str]) -> List[RunPlan]:
    plans: List[RunPlan] = []
    for p in paths:
        n_rows = _count_rows_fast(p)
        avg_b = _estimate_avg_row_bytes(p)
        # read tiny to infer run_id
        df0 = pd.read_csv(p, nrows=5)
        run_id = _infer_run_id(p, df0)
        plans.append(RunPlan(path=p, run_id=run_id, n_rows=n_rows, avg_row_bytes=avg_b))
    return plans


def bundle(
    paths: List[str],
    out_prefix: str,
    max_mb: int,
    min_rows_per_run: int,
    reserve_mb: int = 1,
) -> List[str]:
    """
    Собирает в bundle_partXX.csv, ограничивая размер каждого файла.
    """
    max_bytes = max_mb * 1024 * 1024
    reserve_bytes = reserve_mb * 1024 * 1024
    limit = max(1_000_000, max_bytes - reserve_bytes)

    plans = plan_all_runs(paths)

    # approximate: allocate rows per run proportional to length, based on total budget
    # budget in rows = limit / avg_row_bytes (use median to be robust)
    med_bpr = int(np.median([p.avg_row_bytes for p in plans])) if plans else 200
    total_budget_rows = max(100, int(limit / max(50, med_bpr)))

    total_rows = sum(p.n_rows for p in plans) or 1

    allocations: Dict[str, int] = {}
    for p in plans:
        # proportional allocation, but at least min_rows_per_run
        k = int(total_budget_rows * (p.n_rows / total_rows))
        k = max(min_rows_per_run, k)
        allocations[p.run_id] = k

    # Output parts
    out_files: List[str] = []
    part_idx = 1
    current_path = f"{out_prefix}_part{part_idx:02d}.csv"
    current_bytes = 0
    wrote_header = False
    header_cols: Optional[List[str]] = None

    def _open_new_part():
        nonlocal part_idx, current_path, current_bytes, wrote_header, header_cols
        part_idx += 1
        current_path = f"{out_prefix}_part{part_idx:02d}.csv"
        current_bytes = 0
        wrote_header = False
        # header_cols stays the same

    # iterate runs
    for p in plans:
        df = pd.read_csv(p.path)
        # make sure run_id exists
        if "run_id" not in df.columns:
            df.insert(0, "run_id", p.run_id)
        else:
            # fill missing run_id with inferred
            df["run_id"] = df["run_id"].astype(str).replace({"nan": ""})
            if (df["run_id"] == "").all():
                df["run_id"] = p.run_id
            else:
                df["run_id"] = df["run_id"].where(df["run_id"] != "", p.run_id)

        k = allocations.get(p.run_id, min_rows_per_run)
        dfc = compress_timeseries(df, target_rows=k)

        # Harmonize columns across runs: keep union but stable order.
        if header_cols is None:
            header_cols = list(dfc.columns)
        else:
            # union
            for c in dfc.columns:
                if c not in header_cols:
                    header_cols.append(c)
            # add missing cols in dfc
            for c in header_cols:
                if c not in dfc.columns:
                    dfc[c] = np.nan
            dfc = dfc[header_cols]

        # estimate bytes to write
        csv_text = dfc.to_csv(index=False, header=not wrote_header)
        b = len(csv_text.encode("utf-8"))

        if current_bytes + b > limit and current_bytes > 0:
            # new part
            _open_new_part()
            csv_text = dfc.to_csv(index=False, header=True)
            b = len(csv_text.encode("utf-8"))

        # write append
        mode = "a" if os.path.exists(current_path) else "w"
        with open(current_path, mode, encoding="utf-8", newline="") as f:
            f.write(csv_text)

        current_bytes += b
        wrote_header = True

        if current_path not in out_files:
            out_files.append(current_path)

    return out_files

            S_obj = float(np.mean(st.S[mask])) if hasattr(st, "S") else np.nan

        pi, pj = best_ij

        if pi < 0 or pj < 0:
            pi, pj = N // 2, N // 2

        r_in = 2
        r_out = 5  # кольцо 2..4

            if (self._frame % int(diag_stride)) != 0:
                self._maybe_log()
                return

        # --- period from signal: локальная проба у якоря ---
        if len(self.P_buf) >= int(self.cfg.Wmin):

            sig = np.asarray(self.P_buf, float)

                max_period=int(self.cfg.peak_max_period),
                smooth_win=int(self.cfg.peak_smooth_win),
                prominence=float(self.cfg.peak_prominence),
            )
            self._last_peaks = peaks if peaks is not None else []

            if phi is None or T_inst is None or not np.isfinite(T_inst[-1]):
                self.phi_buf.append(np.nan)
                self.T_buf.append(np.nan)
            else:
                self.phi_buf.append(float(phi[-1]) if np.isfinite(phi[-1]) else np.nan)
                self.T_buf.append(float(T_inst[-1]))
        else:
            self.phi_buf.append(np.nan)
            self.T_buf.append(np.nan)

        # --- freeze stats (optional; stays NaN until enough phi and probes) ---
        self._update_freeze()

        # log
        self._maybe_log()

    def _update_freeze(self):
        if self.probe_pts is None or self.probe_bufs is None:
            self.freeze_mean_buf.append(np.nan)
            self.freeze_min_buf.append(np.nan)
            self.locked_frac_buf.append(np.nan)
            return

        win = int(self.cfg.quad_win)
        phi_g = np.asarray(list(self.phi_buf), float)
        if phi_g.size < win or not np.isfinite(phi_g[-1]):
            self.freeze_mean_buf.append(np.nan)
            self.freeze_min_buf.append(np.nan)
            self.locked_frac_buf.append(np.nan)
            return
        phi_g = phi_g[-win:]

        vars_ = []
        for buf in self.probe_bufs:
            sig = np.asarray(list(buf), float)
            if sig.size < win:
                continue
            sig = sig[-win:]
            _, plv, var = local_phase_via_quadrature(sig, phi_g, win=win)
            if np.isfinite(var):
                vars_.append(float(var))

        if len(vars_) == 0:
            self.freeze_mean_buf.append(np.nan)
            self.freeze_min_buf.append(np.nan)
            self.locked_frac_buf.append(np.nan)
            return

        v = np.asarray(vars_, float)
        self.freeze_mean_buf.append(float(np.mean(v)))
        self.freeze_min_buf.append(float(np.min(v)))
        self.locked_frac_buf.append(float(np.mean(v <= float(self.cfg.freeze_thr))))

# core/model.py
import numpy as np
from core.state import State

class Model:
    def __init__(self, st: State, params):
        self.st = st
        self.p = params

    def time_speed(self, S):
        return 1.0 / (1.0 + self.p.lambda_t * S)

    def update_S(self, E):
        N = self.st.N
        E_level = self.st.E_level
        beta = self.p.beta
        gamma = self.p.gamma

        S_new = beta * (E - E_level) ** 2
        for i in range(N):
            for j in range(N):
                for di, dj in [(-1,0),(1,0),(0,-1),(0,1)]:
                    ni, nj = (i + di) % N, (j + dj) % N
                    S_new[i, j] += gamma * (E[ni, nj] - E_level) ** 2
        return S_new

    def compute_mu(self, E):
        beta = self.p.beta
        E_level = self.st.E_level
        return 2 * beta * (E - E_level)

    def step(self):
        st, p = self.st, self.p
        N = st.N
        E_level = st.E_level

        E_new = st.E.copy()
        st.Jx.fill(0.0)
        st.Jy.fill(0.0)

        st.S[:] = self.update_S(st.E)

        mu = self.compute_mu(st.E)
        st.mu_bar = float(mu.mean())
        mu_eff = mu - st.mu_bar

        v = self.time_speed(st.S)
        st.tau += v

        for i in range(N):
            for j in range(N):
                if st.tau[i, j] < 1.0:
                    continue

                st.tau[i, j] -= 1.0
                sigma = 1.0 / (1.0 + p.alpha * st.S[i, j])

                for di, dj in [(1,0),(0,1)]:
                    ni, nj = (i + di) % N, (j + dj) % N
                    phi = p.kappa * sigma * (mu_eff[i, j] - mu_eff[ni, nj])

                    E_new[i, j] -= phi
                    E_new[ni, nj] += phi

                    if di == 1:
                        st.Jx[i, j] -= phi
                    else:
                        st.Jy[i, j] -= phi

        st.E[:] = np.clip(E_new, 0.0, 2.0 * E_level)
        st.t_global += 1.0

# core/model_fast.py
import numpy as np
from core.state import State


class ModelFast:
    """
    Векторная NumPy-реализация Model.step() без Python-циклов.

    ВАЖНО:
    - делаем максимально близко к текущей логике:
      * S зависит от (E-E_level)^2 + вклад соседей (8-соседей)
      * tau += v(S)
      * активные клетки (tau>=1) совершают перенос энергии
      * перенос делаем по двум направлениям как в текущем коде:
        (di=1,dj=0) и (di=0,dj=1) — то есть "вниз" и "вправо"
    """

    def __init__(self, st: State, params):
        self.st = st
        self.p = params

    def time_speed(self, S):
        return 1.0 / (1.0 + self.p.lambda_t * S)

    def update_S(self, E):
        # Базовая часть
        E0 = self.st.E_level
        beta = self.p.beta
        gamma = self.p.gamma

        d = (E - E0)
        S = beta * d * d

        # 8-соседей (как в твоём update_S с di/dj in [-1,0,1], кроме 0,0)
        for di in (-1, 0, 1):
            for dj in (-1, 0, 1):
                if di == 0 and dj == 0:
                    continue
                En = np.roll(np.roll(E, di, axis=0), dj, axis=1)
                dn = (En - E0)
                S += gamma * dn * dn

        return S

    def step(self):
        st = self.st
        p = self.p
        E = st.E

        # 1) S
        S_new = self.update_S(E)
        st.S[:] = S_new

        # 2) mu и глобальная компенсация
        E0 = st.E_level
        beta = p.beta
        mu = 2.0 * beta * (E - E0)
        mu_bar = float(np.mean(mu))
        st.mu_bar = mu_bar
        mu_eff = mu - mu_bar

        # 3) время
        v = self.time_speed(S_new)
        st.tau += v

        # 4) активные клетки
        active = (st.tau >= 1.0)
        st.tau[active] -= 1.0

        # 5) sigma
        # (по твоему смыслу: чем больше энтропия, тем меньше проводимость)
        sigma = np.zeros_like(E, dtype=float)
        sigma[active] = 1.0 / (1.0 + p.alpha * S_new[active])

        # 6) перенос энергии (вниз и вправо) — векторно
        # вниз = (i+1,j): roll(-1, axis=0) даёт E_down[i,j] = E[i+1,j]
        mu_down = np.roll(mu_eff, -1, axis=0)
        mu_right = np.roll(mu_eff, -1, axis=1)

        # phi = kappa * sigma * (mu_eff - mu_neighbor)
        phi_x = p.kappa * sigma * (mu_eff - mu_down)   # поток "вниз"
        phi_y = p.kappa * sigma * (mu_eff - mu_right)  # поток "вправо"

        # Jx/Jy по знаку как у тебя: ты делаешь st.Jx[i,j] -= phi, st.Jy[i,j] -= phi
        st.Jx[:] = -phi_x
        st.Jy[:] = -phi_y

        # E_new = E - out + in
        E_new = E - phi_x - phi_y
        E_new += np.roll(phi_x, 1, axis=0)  # пришло сверху
        E_new += np.roll(phi_y, 1, axis=1)  # пришло слева

        st.E[:] = np.clip(E_new, 0.0, 2.0 * E0)
        st.t_global += 1.0

from __future__ import annotations

try:
    import torch
    import torch.nn.functional as F
except Exception:  # torch may be absent
    torch = None
    F = None


class TorchBackendError(RuntimeError):
    pass


def require_torch():
    if torch is None:
        raise TorchBackendError(
            "PyTorch is not installed. Install it, e.g.:\n"
            "  pip install torch\n"
            "or CUDA build:\n"
            "  pip install torch --index-url https://download.pytorch.org/whl/cu121"
        )


class ModelTorchFast:
    """
    Torch-реализация шага, максимально близкая к core/model_fast.py.
    Работает на CPU/GPU. Опционально можно обернуть в torch.compile в вызывающем коде.
    """

    def __init__(self, st, params):
        require_torch()
        self.st = st
        self.p = params

        # заранее подготовим 3x3 ядро "соседи кроме центра"
        # Оно будет использоваться для суммы (E-E0)^2 по 8 соседям
        k = torch.ones((1, 1, 3, 3), device=st.E.device, dtype=st.E.dtype)
        k[0, 0, 1, 1] = 0.0
        self._k8 = k

    def time_speed(self, S):
        return 1.0 / (1.0 + float(self.p.lambda_t) * S)

    def update_S(self, E):
        """
        Аналог ModelFast.update_S:
          S = beta*(E-E0)^2 + gamma*sum_8neighbors( (E_neighbor-E0)^2 )
        Важно: в numpy-версии соседи берутся с периодическими границами (np.roll).
        Тут делаем circular padding + conv2d.
        """
        E0 = float(self.st.E_level)
        beta = float(self.p.beta)
        gamma = float(self.p.gamma)

        d = (E - E0)
        base = beta * d * d

        # conv по (d^2) с circular padding
        # F.pad поддерживает mode="circular"
        x = (d * d)[None, None, :, :]          # [1,1,N,N]
        x = F.pad(x, (1, 1, 1, 1), mode="circular")
        neigh_sum = F.conv2d(x, self._k8)[0, 0]  # [N,N]

        return base + gamma * neigh_sum

    @torch.no_grad()
    def step(self):
        st = self.st
        p = self.p

        E = st.E

        # 1) S
        S_new = self.update_S(E)
        st.S.copy_(S_new)

        # 2) mu и глобальная компенсация
        E0 = float(st.E_level)
        beta = float(p.beta)

        mu = 2.0 * beta * (E - E0)
        mu_bar = float(mu.mean().item())
        st.mu_bar = mu_bar
        mu_eff = mu - mu_bar

        # 3) время
        v = self.time_speed(S_new)
        st.tau.add_(v)

        # 4) активные клетки
        active = (st.tau >= 1.0)
        st.tau[active] -= 1.0

        # 5) sigma
        sigma = torch.zeros_like(E)
        alpha = float(p.alpha)
        sigma[active] = 1.0 / (1.0 + alpha * S_new[active])

        # 6) перенос энергии (вниз и вправо) — как ModelFast
        kappa = float(p.kappa)

        mu_down = torch.roll(mu_eff, shifts=-1, dims=0)
        mu_right = torch.roll(mu_eff, shifts=-1, dims=1)

        phi_x = kappa * sigma * (mu_eff - mu_down)   # вниз
        phi_y = kappa * sigma * (mu_eff - mu_right)  # вправо

        st.Jx.copy_(-phi_x)
        st.Jy.copy_(-phi_y)

        E_new = E - phi_x - phi_y
        E_new = E_new + torch.roll(phi_x, shifts=1, dims=0)
        E_new = E_new + torch.roll(phi_y, shifts=1, dims=1)

        # clip как в numpy fast: [0, 2*E0]
        E_new = torch.clamp(E_new, 0.0, 2.0 * E0)
        st.E.copy_(E_new)

        st.t_global = float(st.t_global) + 1.0

def div2d(Jx, Jy):
    # axis0 = rows (x), axis1 = cols (y) — as in your original code
    dJx = 0.5 * (np.roll(Jx, -1, axis=0) - np.roll(Jx, 1, axis=0))
    dJy = 0.5 * (np.roll(Jy, -1, axis=1) - np.roll(Jy, 1, axis=1))
    return dJx + dJy


def curl2d(Jx, Jy):
    dJy_dx = 0.5 * (np.roll(Jy, -1, axis=0) - np.roll(Jy, 1, axis=0))
    dJx_dy = 0.5 * (np.roll(Jx, -1, axis=1) - np.roll(Jx, 1, axis=1))
    return dJy_dx - dJx_dy


def rings_indices(N, center, r_max=None):
    cx, cy = center
    xs = np.arange(N)[:, None]
    ys = np.arange(N)[None, :]
    rr = np.sqrt((xs - cx) ** 2 + (ys - cy) ** 2)
    if r_max is None:
        r_max = int(rr.max())
    ridx = np.clip(rr.astype(int), 0, r_max)
    return ridx, r_max


def ring_means(field, ridx, r_max):
    # field: [T,N,N] or [N,N]
    if field.ndim == 2:
        out = np.zeros((r_max + 1,), dtype=np.float64)
        for r in range(r_max + 1):
            m = (ridx == r)
            out[r] = field[m].mean() if m.any() else np.nan
        return out
    else:
        T = field.shape[0]
        out = np.zeros((T, r_max + 1), dtype=np.float64)
        for r in range(r_max + 1):
            m = (ridx == r)
            if not m.any():
                out[:, r] = np.nan
            else:
                out[:, r] = field[:, m].mean(axis=1)
        return out

def quantize_linear_int8(x, clip_sigma=4.0):
    """
    Robust int8 quantization around mean/std.
    Returns (q:int8, scale:float32, zero:float32).
    Reconstruct approx: x_hat = scale*q + zero
    """
    x = x.astype(np.float32, copy=False)
    mu = np.nanmean(x, dtype=np.float64)
    sig = np.nanstd(x, dtype=np.float64)
    if not np.isfinite(sig) or sig < 1e-12:
        sig = 1.0
    lo = mu - clip_sigma * sig
    hi = mu + clip_sigma * sig
    x_clip = np.clip(x, lo, hi)

    scale = (hi - lo) / 254.0
    if scale < 1e-12:
        scale = 1.0

    mid = (lo + hi) / 2.0
    q = np.round((x_clip - mid) / scale).astype(np.int16)
    q = np.clip(q, -127, 127).astype(np.int8)

    return q, np.float32(scale), np.float32(mid)

def div2d(Jx, Jy):
    # NOTE: axis convention must match your field layout.
    # Here assumes axis0 = x (rows), axis1 = y (cols), consistent with your original code.
    dJx = 0.5 * (np.roll(Jx, -1, axis=0) - np.roll(Jx, 1, axis=0))
    dJy = 0.5 * (np.roll(Jy, -1, axis=1) - np.roll(Jy, 1, axis=1))
    return dJx + dJy


def curl2d(Jx, Jy):
    dJy_dx = 0.5 * (np.roll(Jy, -1, axis=0) - np.roll(Jy, 1, axis=0))
    dJx_dy = 0.5 * (np.roll(Jx, -1, axis=1) - np.roll(Jx, 1, axis=1))
    return dJy_dx - dJx_dy


def rings_indices(N, center, r_max=None):
    cx, cy = center
    xs = np.arange(N)[:, None]
    ys = np.arange(N)[None, :]
    rr = np.sqrt((xs - cx) ** 2 + (ys - cy) ** 2)
    if r_max is None:
        r_max = int(rr.max())
    ridx = np.clip(rr.astype(int), 0, r_max)
    return ridx, r_max


def ring_means(field, ridx, r_max):
    # field: [T,N,N] or [N,N]
    if field.ndim == 2:
        out = np.zeros((r_max + 1,), dtype=np.float64)
        for r in range(r_max + 1):
            m = (ridx == r)
            out[r] = field[m].mean() if m.any() else np.nan
        return out
    else:
        T = field.shape[0]
        out = np.zeros((T, r_max + 1), dtype=np.float64)
        for r in range(r_max + 1):
            m = (ridx == r)
            if not m.any():
                out[:, r] = np.nan
            else:
                out[:, r] = field[:, m].mean(axis=1)
        return out


def find_hist_key(d, candidates):

    return None


def quantize_linear_int8(x, clip_sigma=4.0):
    """
    Simple robust int8 quantization around mean/std.
    Returns (q:int8, scale:float32, zero:float32).
    Reconstruct approx: x_hat = scale*(q) + zero
    """
    x = x.astype(np.float32, copy=False)
    mu = np.nanmean(x, dtype=np.float64)
    sig = np.nanstd(x, dtype=np.float64)
    if not np.isfinite(sig) or sig < 1e-12:
        sig = 1.0
    lo = mu - clip_sigma * sig
    hi = mu + clip_sigma * sig
    x_clip = np.clip(x, lo, hi)
    # map [lo,hi] -> [-127,127]
    scale = (hi - lo) / 254.0
    if scale < 1e-12:
        scale = 1.0
    q = np.round((x_clip - (lo + hi) / 2.0) / scale).astype(np.int16)
    q = np.clip(q, -127, 127).astype(np.int8)
    zero = np.float32((lo + hi) / 2.0)
    scale = np.float32(scale)
    return q, scale, zero

## 4. Иерархия уровней (L → L+1)

### 4.1 Принцип

> **Узел уровня L является точкой на уровне L+1.**

Переход на уровень выше — это **коарсинг**:

- пространственный масштаб ↑  
- временной отклик ↑  
- точность ↓  
- устойчивость ↑  

Следствие:

- большие объекты имеют большее внутреннее время;  
- верхние уровни реагируют медленнее, но на больший контекст.

## 5. Асимметрия знания между уровнями (инвариант)

### 5.1 Что видит уровень L+1

Уровень L+1 **не видит**:

- внутреннюю структуру уровня L;  
- решётку;  
- микродинамику.

Он видит только:

- точку;  
- её **эффективное действие**;  
- параметры среды этой точки.

### 5.2 Что видит уровень L

Точка уровня L:

- не знает, что она точка;  
- не знает о существовании L+1;  
- эволюционирует только под действием:
  - внешнего поля;  
  - собственных параметров среды.

> ⛔ Любая обратная “осознанность” запрещена.

## 6. Межуровневые события: информация и управление

Межуровневые связи **не являются переносом энергии или материи**.

Реально происходит:

- событие возникает на уровне L;  
- на уровнях L+1…L+k распространяется **интерпретация события**.

При подъёме по уровням:

- точность ↓  
- устойчивость ↑  
- длительность ↑  
- радиус контекста ↑

## 7. Определение смысла (операционально)

> **Смысл = способность события пережить подъём по уровням.**

Пусть `L_max(e)` — максимальный уровень, где событие остаётся значимым:

- шум → исчезает на L+1;  
- паттерн → живёт до L+5;  
- концепт → до L+20;  
- цель / мотивация → L+50+.

Смысл — **измеримая устойчивость при коарсинге**.

## 8. Patch–fallback (пересчёт при переполнении ёмкости)

Каждый нейрон уровня L имеет **ёмкость**.

Если состояние выходит за basin of attraction:

1. вырезается patch уровня L−1;  
2. пересчитывается детальной моделью;  
3. окружение взаимодействует через граничные потоки;  
4. результат агрегируется обратно в нейрон L.

Это обеспечивает:

- точность при необходимости;  
- O(1)-работу в стабильных режимах.

---
## 8A. Альтернатива Patch–fallback: Fallback как вложенная симуляция (pipeline swap)

### 8A.1 Идея

Вместо “вырезания патча внутри общего поля” fallback трактуется как **подмена пайплайна вычисления одной ячейки/нейрона** уровня $\mathrm{L}$:

- в стабильном режиме ячейка уровня $\mathrm{L}$ реализуется как **O(1)-нейрон** (инварианты + локальный оператор отклика);
- при переполнении ёмкости (выход из basin of attraction) ячейка временно заменяется на **вложенный subworld** — отдельную симуляцию уровня $\mathrm{L-1}$ … $\mathrm{L-k}$ (обычно $k \le 3$).

Таким образом, fallback создаёт **отдельное поле** для одной ячейки, а не “встраиваемый патч” в глобальную решётку.

---

### 8A.2 Единственный канал связи: граничный интерфейс

Связь внешней среды (уровня $\mathrm{L}$) и subworld (уровня $\mathrm{L-1}$ …) осуществляется **только через границу**.

Внешняя среда задаёт subworld **входное воздействие**:

$$
B_{\mathrm{in}}(r,t)
=
\mathcal{B}_{\mathrm{in}}
\Big(
\text{окрестность } r \text{ на уровне } \mathrm{L},
\Theta_{\mathrm{L}}(r),
t
\Big)
$$

Subworld отдаёт наружу **интегральный отклик**:

$$
B_{\mathrm{out}}(r,t)
=
\mathcal{B}_{\mathrm{out}}
\Big(
\text{состояние subworld},
t
\Big)
$$

Интерпретация:

- $B_{\mathrm{in}}$ играет роль граничных условий (внешнее поле/потоки/дефицит);
- $B_{\mathrm{out}}$ — это “эффективное действие точки” на уровень $\mathrm{L}$, т.е. то, что соседние клетки видят как влияние данной клетки.

---

### 8A.3 Нормировка и каузальность

Чтобы исключить “энергию из воздуха” и усиление вверх:

- $B_{\mathrm{out}}$ обязан быть **нормированным** и согласованным с инвариантом $E_{\mathrm{L}} \in [0,1]$;
- допускается сглаживание отклика во времени для устранения дрожания:

$$
B_{\mathrm{out}}^{\mathrm{smooth}}(t)
=
(1-\rho)\,B_{\mathrm{out}}^{\mathrm{smooth}}(t-1)
+
\rho\,B_{\mathrm{out}}(t)
$$

где $\rho \in (0,1]$.

---

### 8A.4 Планировщик subworld (ограничение ресурса)

Поскольку subworld является отдельной симуляцией, вычисления можно планировать:

- вводится максимум одновременно активных subworld:
  $$
  M = \max\ \text{числа subworld, считаемых параллельно}
  $$
- остальные subworld ставятся в очередь и обслуживаются по приоритету (например, по амплитуде/новизне/влиянию).

Ресурсная стоимость определяется не “числом патчей на карте”, а:

- числом активных subworld $M(t)$,
- их временем жизни,
- глубиной разворота $k$ (обычно $k \le 3$).

---

### 8A.5 Критерий свёртки subworld обратно в нейрон

Subworld сворачивается обратно в O(1)-нейрон уровня $\mathrm{L}$, когда:

- параметры инвариантов стабилизировались,
- частота/амплитуда требуемых уточнений снизилась,
- отклик $B_{\mathrm{out}}(r,t)$ стал устойчивым при фиксированном $B_{\mathrm{in}}$.

---

### 8A.6 Эквивалентность по смыслу Patch–fallback

Эта реализация сохраняет смысл исходного fallback:

- точность при необходимости (локальная детализация),
- O(1)-работа в стабильных режимах,
- отсутствие прямого переноса энергии между уровнями,
- межуровневые связи реализуются как **управление условиями эволюции** через граничный интерфейс.

Разница только техническая:

- Patch–fallback: “встроенная детализация в рамках общей решётки”;
- Pipeline swap: “отдельный вложенный мир с граничным I/O”.

## 9A. Нормировка энергии и шкала уровней (канонический инвариант)

### 9A.1 Основной инвариант энергии

Во всей модели вводится жёсткое ограничение:

Энергия **никогда не превышает 1** ни на одном уровне и не “усиливается” при переходе между уровнями.

Рост масштаба выражается **не через увеличение $E$**, а через:
- изменение меры объёма,
- изменение масштаба уровня,
- изменение контекста и времени реакции.

### 9A.2 Разделение нормированной энергии и физической меры

В модели строго разделяются:

1. **Нормированное состояние** (участвует в динамике):
   \[
   E_{\mathrm{L}}(r) \in [0,1]
   \]

2. **Масштабная мера уровня** (не участвует в локальной динамике):
   \[
   \mu_{\mathrm{L}} \sim s^{d\mathrm{L}}
   \]
   где:
   - $s$ — коэффициент масштаба (по умолчанию $s = 10$),

Таким образом, при подъёме по уровням растёт **не энергия клетки**, а **мера пространства**, которое она представляет.

### 9A.3 Физическая энергия как интерпретация (не динамика)

где:

**Важно:**  
$E_{\mathrm{phys}}$ **не участвует в динамике** и используется только для:
- интерпретации,

### 9A.4 Канонический оператор коарсинга энергии

Переход от уровня $\mathrm{L}$ к $\mathrm{L+1}$ **не является суммированием** энергий.

Для патча $P_{\mathrm{L}}(r)$ используется нормирующий агрегатор.

Рекомендуемые формы:

#### A) Насыщаемая агрегация
\[
E_{\mathrm{L+1}}(r)
=
1 -
\prod_{x \in P_{\mathrm{L}}(r)}
\bigl(1 - E_{\mathrm{L}}(x)\bigr)
\]

#### B) Логит-нормализация (предпочтительно)
\[
u = \log \frac{E}{1-E},
\qquad
u_{\mathrm{L+1}}(r)
=
\frac{1}{|P|}
\sum_{x \in P_{\mathrm{L}}(r)} u_{\mathrm{L}}(x),
\qquad
E_{\mathrm{L+1}}(r)
=
\frac{1}{1 + e^{-u_{\mathrm{L+1}}(r)}}
\]

Оба варианта гарантируют:
- $E_{\mathrm{L+1}} \in [0,1]$,
- отсутствие переполнений,
- сохранение локализации и границ объектов.

Уровень $\mathrm{L}$ трактуется как **координата на непрерывной шкале**, а не только целый индекс.

- $\mathrm{L} = 0.52$
- $\mathrm{L} = 0.643$
- $\mathrm{L} = 532.463$

Это позволяет:
- естественно описывать промежуточные масштабы,
- избегать жёстких дискретных скачков,
- использовать представление вида:
  \[
  E_{\mathrm{phys}} = E_{\mathrm{norm}} \cdot s^{\mathrm{L}}
  \]

### 9A.6 Вычислительная мотивация

Такое разделение даёт:

- фиксированный диапазон значений $E \in [0,1]$ на всех уровнях;

Рост сложности системы выражается:

но **не в росте амплитуд локальных величин**.

## 10A. Автоматизация обучения уровней и критерий сложности (канонический контур)

### 10A.1 Цель контура

Система рассматривается не только как симулятор, но как механизм **добычи инвариантов**:

1. Модель эволюционирует на уровне $\mathrm{L}$ до устойчивого режима.
2. Устойчивый режим изучается (диагностика) и сворачивается в O(1)-представление.
3. Это представление фиксируется как **нейрон уровня $\mathrm{L}$** (набор инвариантов).
4. Строится уровень $\mathrm{L+1}$ как решётка нейронов $\mathrm{L}$ с теми же общими законами взаимодействия.
5. Интерес пользователя обычно ограничен 2–3 уровнями вверх, но вычислительная сложность определяется провалами вниз (fallback).

---

### 10A.2 Принцип “нейрон готов” (закрытие уровня)

Нейрон уровня $\mathrm{L}$ считается сформированным, когда:

- параметры устойчивого аттрактора стабилизировались,
- провалы вниз стали редкими и мелкими,
- макродинамика на уровне $\mathrm{L}$ описывается O(1) параметрами.

Формально вводится вектор параметров (инвариантов нейрона):
\[
\Theta_{\mathrm{L}} = \{\text{центр}, \text{масштаб}, \text{ориентация/ширина}, \text{период}, \dots\}
\]

и критерий стабилизации:
\[
\mathrm{Var}\bigl(\Theta_{\mathrm{L}}\bigr) < \theta_{\Theta}
\]

---

### 10A.3 Ресурсная сложность как функция fallback (основной критерий)

Основная вычислительная стоимость определяется не размером пространства, а числом и глубиной провалов вниз.

Обозначим:

- $P(t)$ — число активных патчей (fallback) во времени,
- $D(t)$ — средняя глубина провала вниз (в уровнях),
- $N_{\mathrm{active}}(\mathrm{L})$ — число активных нейронов на уровне $\mathrm{L}$,
- $f_{\mathrm{act}}(\mathrm{L})$ — частота активации уровня $\mathrm{L}$.

Вводится “интенсивность провалов”:
\[
\mathrm{Fall}(T)
=
\frac{1}{T}\sum_{t=1}^{T} P(t)\,D(t)
\]

и “индекс нестабильности” уровня:
\[
\mathrm{Instab}(\mathrm{L};T)
=
\frac{1}{T}\sum_{t=1}^{T}
\frac{P_{\mathrm{L}}(t)}{N_{\mathrm{active}}(\mathrm{L})+\varepsilon}
\]

Интерпретация:

- большие значения $\mathrm{Fall}(T)$ и $\mathrm{Instab}(\mathrm{L};T)$ означают, что уровень $\mathrm{L}$ плохо описывается O(1)-нейронами и часто требует детализации;
- это является сигналом, что система находится в “нестабильной” области пространства параметров или в неудачном режиме.

---

### 10A.4 Критерий остановки и смены режима поиска

Если в течение окна $T$:

\[
\mathrm{Fall}(T) > \theta_{\mathrm{fall}}
\quad\text{или}\quad
\mathrm{Instab}(\mathrm{L};T) > \theta_{\mathrm{instab}}
\]

то считается, что:

- текущий режим параметров/условий приводит к чрезмерным провалам вниз,
- вычислительная сложность растёт,
- и целесообразно:
  - изменить параметры среды,
  - изменить начальные условия,
  - или сменить область поиска.

Это реализует практическую эвристику:

> “Если система много раз возвращается на нижние уровни, значит она нестабильна; возможно, стоит искать другой режим.”

---

### 10A.5 Замечание об обучении и диагностике

Контур “добычи инвариантов” должен реализовываться через внешнюю диагностику:

- диагностика читает immutable snapshots;
- динамика не “подстраивается” под диагностику;
- фиксация нейрона уровня $\mathrm{L}$ — это отдельная операция, не вмешивающаяся в симуляцию.

## 11. Ограничения

- модель не является ОТО или КМ;  
- энергия **не усиливается** вверх по уровням;  
- все гипотезы проверяются экспериментально;  
- телепатия и скрытая синхронизация запрещены.

## 11A. Форматы хранения нейронов и subworld (SSD / RAM / GPU)

### 11A.1 Общий принцип хранения

Архитектура жёстко разделяет:

- **хранение** (SSD) — долговременная библиотека инвариантов;
- **исполнение** (RAM / GPU) — только активные нейроны и subworld;
- **кэш** (RAM) — горячие нейроны и недавно используемые subworld.

Ни один уровень не предполагает постоянного хранения полных решёток всех масштабов в памяти.

---

### 11A.2 Формат нейрона уровня $\mathrm{L}$ (SSD, канонический)

Нейрон уровня $\mathrm{L}$ хранится как **самодостаточный O(1)-объект**, не зависящий от глобальной решётки.

#### Минимальный состав нейрона

| Поле | Назначение |
|----|----|
| `neuron_id` | Уникальный идентификатор (hash) |
| `level_L` | Уровень $\mathrm{L}$ |
| `Theta_L` | Параметры аттрактора (центр, масштаб, ориентация, период, …) |
| `inv_topK` | Top-$K$ инвариантов (float32 / float16) |
| `inv_scores` | Значимость / устойчивость инвариантов |
| `stability_stats` | Статистика стабилизации и fallback |
| `boundary_model` | Параметры оператора $B_{\mathrm{out}}$ |
| `meta` | Версия, хэш правил, дата, источник |

#### Типичные размеры

- $K = 256$ → ~1–2 КБ
- $K = 1024$ → ~4–8 КБ
- с метаданными и статистикой: **16–64 КБ на нейрон** (реалистично)

---

### 11A.3 Формат subworld (временный объект)

Subworld **не является постоянной сущностью** и не хранится в библиотеке как “объект уровня”.

Он создаётся, живёт и уничтожается.

#### Состав subworld

| Компонент | Назначение |
|---------|------------|
| `subworld_id` | Временный ID |
| `parent_neuron_id` | Нейрон уровня $\mathrm{L}$ |
| `depth_k` | Глубина разворота ($\mathrm{L-k}$) |
| `grid_shape` | Размер решётки (например $1000\times1000$) |
| `fields` | $E, S, \tau, \vec J$ (или минимальный набор) |
| `boundary_state` | Текущее $B_{\mathrm{in}}$ |
| `runtime_stats` | Время жизни, шаги, стоимость |

Subworld **никогда не сериализуется целиком** на SSD — сохраняется только результат свёртки.

---

### 11A.4 Свёртка subworld → нейрон

По завершении subworld:

1. вычисляется устойчивый интегральный отклик:
   $$
   B_{\mathrm{out}}(t)
   =
   \mathcal{B}_{\mathrm{out}}
   (\text{subworld})
   $$
2. извлекаются или уточняются параметры $\Theta_{\mathrm{L}}$;
3. обновляются статистики устойчивости;
4. формируется **новая версия нейрона** или обновляется существующая.

Результат сохраняется на SSD как **новый или обновлённый нейрон**.

---

### 11A.5 Иерархия хранения и кэширование

#### Уровни хранения

| Уровень | Назначение |
|------|-----------|
| SSD | Полная библиотека нейронов |
| RAM | Кэш горячих нейронов + активные subworld |
| GPU | Только активные subworld и рабочие буфера |

#### Кэш RAM (рекомендуемо)

- LRU / LFU кэш нейронов;
- приоритет:
  - нейроны с высокой частотой fallback,
  - нейроны, активные в текущем окне времени.

---

### 11A.6 Ограничение ресурса и планирование

Вводятся явные лимиты:

- $M_{\mathrm{GPU}}$ — максимум subworld на GPU;
- $M_{\mathrm{RAM}}$ — максимум subworld в RAM;
- очередь subworld обслуживается по приоритету:
  - амплитуда возмущения,
  - влияние на соседей,
  - новизна режима.

Это обеспечивает:

- контролируемое потребление памяти,
- отсутствие лавинообразного роста стоимости,
- предсказуемость времени шага.

---

### 11A.7 Метрика стоимости хранения и вычислений

Для диагностики и обучения логируются:

- число активных subworld $M(t)$,
- среднее время жизни subworld,
- глубина разворота $k$,
- частота обновления нейронов,
- объём SSD библиотеки по уровням.

Эти величины используются для:
- оценки сложности режима,
- выбора параметров среды,
- принятия решения о смене области поиска.

---

### 11A.8 Канонический запрет

> **Запрещено**:
> - хранить полные решётки subworld как постоянные объекты;
> - передавать состояние subworld напрямую соседям;
> - смешивать библиотеку нейронов и рабочие поля.

Нейрон — это инвариант.  
Subworld — это временный вычислительный инструмент.

---
## 12A. Два времени в модели: глобальный тик и собственное время объектов

### 12A.1 Разделение понятий времени

В модели вводится принципиальное различие между двумя временными величинами:

1. **Глобальный тик** $t$  
   Дискретный каузальный счётчик шагов вычисления.
   - всегда монотонно возрастает;
   - одинаков для всей симуляции;
   - задаёт порядок применения правил динамики;
   - не является наблюдаемым “физическим временем”.

2. **Собственное время объекта** $\tau_{\mathrm{obj}}$  
   Эмерджентная характеристика устойчивого объекта, определяемая через его
   **внутренний период или цикл**.

Глобальный тик — это “ось причинности”.  
Собственное время — это “часы объекта”.

---

### 12A.2 Операциональное определение собственного времени

Собственное время объекта определяется **не напрямую**, а через измеряемые характеристики:

- устойчивый период колебаний энергии;
- период фазового цикла;
- характерное время релаксации;
- период повторяемости макросостояния.

Формально:

$$
\tau_{\mathrm{obj}}
=
T_{\mathrm{cycle}}
=
\text{число глобальных тиков, необходимых для завершения одного устойчивого цикла}
$$

Таким образом, время измеряется **через счёт повторяемых событий**, а не как абстрактный параметр.

---

### 12A.3 Зависимость собственного времени от масштаба

Эмпирически наблюдается:

- устойчивые объекты имеют собственный период;
- период возрастает с ростом размера объекта;
- внутри объекта собственное время стабилизируется.

Это объясняется механически:

- перенос энергии локален;
- скорость переноса ограничена (каузальность);
- эффективная скорость подавляется энтропией и вязкостью среды.

Характерное время объекта порядка:

$$
\tau_{\mathrm{obj}}
\sim
\frac{R}{v_{\mathrm{eff}}}
$$

где:
- $R$ — характерный размер объекта;
- $v_{\mathrm{eff}}$ — эффективная скорость перестройки состояния
  (зависит от энтропии $S$ и локального внутреннего времени).

Следствие:

> **Чем больше объект, тем медленнее идут его “часы”.**

---

### 12A.4 Время как характеристика среды

В данной модели время трактуется не как фундаментальная субстанция,
а как **следствие свойств среды**.

Интерпретация:

- среда задаёт допустимую скорость перестройки состояний;
- эта скорость определяет, сколько глобальных тиков требуется для реализации цикла;
- собственное время объекта — это количественная мера этой скорости.

Иными словами:

> **Время — это мера пропускной способности среды для устойчивой динамики.**

---

### 12A.5 Роль глобального тика

Глобальный тик выполняет служебную роль:

- обеспечивает каузальный порядок;
- синхронизирует обновления;
- позволяет связывать асинхронные процессы.

Он не наблюдается как “время” внутри модели, а служит универсальным индексом вычисления.

---

### 12A.6 Масштабирование времени при коарсинге уровней

При переходе от уровня $\mathrm{L}$ к уровню $\mathrm{L+1}$:

- пространственный масштаб увеличивается;
- характерные времена объектов возрастают;
- собственное время становится более инерционным.

Формально:

$$
\tau_{\mathrm{L+1}}
\approx
\tau_{\mathrm{L}} \cdot s^{\beta}
$$

где:
- $s$ — коэффициент коарсинга;
- $\beta \approx 1$ — эмпирический показатель.

Это согласуется с наблюдением, что верхние уровни:
- реагируют медленнее,
- но более устойчивы,
- и интегрируют больший контекст.

---

### 12A.7 Связь с асинхронными subworld

В архитектуре с pipeline-swap:

- subworld обладают собственным микровременем;
- они эволюционируют асинхронно относительно глобального тика;
- наружу они публикуют сглаженный отклик.

Это реализует практическое разделение:

- **тик** — синхронизация;
- **время** — внутренняя динамика.

---

### 12A.8 Экспериментальные метрики для проверки гипотезы

Для подтверждения масштабируемости времени рекомендуется логировать:

1. **Период объекта** $T_{\mathrm{cycle}}$
2. **Размер объекта** $R$
3. **Среднюю энтропию внутри объекта** $\langle S \rangle$
4. **Среднее внутреннее время** $\langle \tau \rangle$
5. **Скорость распространения возмущений** $v_{\mathrm{front}}$

Проверяемые зависимости:

$$
T_{\mathrm{cycle}} \propto R^{\alpha}
$$

$$
T_{\mathrm{cycle}} \uparrow \quad \text{при} \quad \langle S \rangle \uparrow
$$

$$
T_{\mathrm{cycle}} \sim \frac{R}{v_{\mathrm{front}}}
$$

Совпадение этих зависимостей по порядку величин подтверждает,
что собственное время является эмерджентным свойством среды.

---

### 12A.9 Итоговая формулировка (каноническая)

> В модели отсутствует фундаментальное “время” как физическая сущность.  
> Существует глобальный каузальный тик и эмерджентное собственное время объектов.  
> Собственное время определяется скоростью устойчивой динамики,
> которая, в свою очередь, задаётся свойствами среды и масштабом объекта.

## 13A. Language / LLM-mode: обучение через внешние возмущения и устойчивые ответы

### 13A.1 Общая идея

Модель может работать в режиме, аналогичном LLM, если:

- вход задаётся как **внешнее воздействие** на среду;
- скрытое состояние реализуется через **устойчивые объекты и их инварианты**;
- выход — это **ответ среды** на воздействие;
- обучение сводится к согласованию:
  > «какое воздействие → какой устойчивый ответ → какое следующее воздействие».

В этом режиме модель не аппроксимирует язык напрямую,  
а **реализует предсказание следующего события через динамику среды**.

---

### 13A.2 Роль глобального тика

В режиме LLM:

- **global tick** задаёт дискретный шаг подачи последовательности;
- каждый тик соответствует одному входному элементу (токену);
- динамика среды между тиками может быть асинхронной и многошаговой.

Global tick играет роль временной оси последовательности,
но не является “временем языка”.

---

### 13A.3 Токен как внешнее возмущение (Token → Injection)

Каждый токен $x_t$ кодируется не как символ,
а как **нормированное возмущение поля**.

Вводится оператор инъекции:

$$
\Delta E(r,t) = \mathcal{I}(x_t, r;\,\phi)
$$

где:
- $\mathcal{I}$ — фиксированный или обучаемый оператор кодирования;
- $r$ — координаты ячеек в области инъекции;
- $\phi$ — параметры кодировщика (при необходимости).

Инъекция:
- локальна (например, область $k \times k$);
- ограничена по энергии;
- не нарушает инвариант $E \in [0,1]$.

Пример нормированной инъекции:

$$
E(r,t^+) = 1 - (1 - E(r,t))(1 - \Delta E(r,t))
$$

Таким образом, токен — это **контролируемый импульс**, а не “запись состояния”.

---

### 13A.4 Объект как ответ среды

После подачи инъекции система эволюционирует
(внутренние тики, асинхронные subworld).

Ответ среды описывается через **инварианты и параметры устойчивых объектов**.

Вводится оператор считывания:

$$
y_t = \mathcal{R}\big(\text{state}_{t:t+\Delta}\big) \in \mathbb{R}^K
$$

где:
- $y_t$ — вектор активностей инвариантов уровня $\mathrm{L}$;
- $K$ — число выбранных инвариантов (top-$K$);
- $\mathcal{R}$ — оператор агрегации (энергия, потоки, частоты, устойчивость).

$y_t$ является **аналогом hidden state** в классическом LLM.

---

### 13A.5 Предсказание следующего токена (Readout)

Предсказание формируется обычной линейной “головой”:

$$
p(x_{t+1} \mid \cdot)
=
\mathrm{softmax}(W y_t + b)
$$

где:
- $W, b$ — обучаемые параметры;
- softmax определяет распределение по словарю.

Таким образом:
- динамика среды → формирует представление;
- обучение сводится к настройке лёгкой головы и (опционально) инъекции.

---

### 13A.6 Функция потерь

Базовая функция потерь — стандартная cross-entropy:

$$
\mathcal{L}_{\mathrm{CE}}
=
-\log p(x_{t+1}^{\mathrm{true}})
$$

Дополнительно могут вводиться регуляризаторы:

- штраф за избыточный fallback:
  $$
  \mathcal{L}_{\mathrm{fb}} \propto \text{cost}_{\mathrm{subworld}}
  $$
- штраф за хаотичность (низкую устойчивость ответа);
- поощрение устойчивых режимов (долгая жизнь инвариантов).

Итог:

$$
\mathcal{L}
=
\mathcal{L}_{\mathrm{CE}}
+
\lambda_{\mathrm{fb}}\mathcal{L}_{\mathrm{fb}}
+
\lambda_{\mathrm{stab}}\mathcal{L}_{\mathrm{stab}}
$$

---

### 13A.7 Иерархия размерностей (N → N+1)

Интерпретация уровней:

- объект уровня $\mathrm{L}$ является точкой для уровня $\mathrm{L+1}$;
- воздействия уровня $\mathrm{L+1}$ действуют как **интегральное давление** на объект $\mathrm{L}$.

В LLM-mode это трактуется так:

- символ (1D) → возмущение поверхности (2D);
- последовательность → серия возмущений;
- смысл → устойчивость ответа при подъёме уровней.

Формально, входное воздействие для объекта уровня $\mathrm{L}$:

$$
B_{\mathrm{in}}^{(\mathrm{L})}(t)
=
\sum_{r \in \partial \Omega_{\mathrm{L}}}
w(r)\,\Psi(r,t)
$$

То есть токен действует не “внутрь”, а **через поверхность/контекст**.

---

### 13A.8 Контекст и внимание (без attention-механизма)

Роль attention выполняют:

- радиус влияния события;
- длительность жизни инварианта;
- устойчивость при коарсинге уровней.

Контекст — это не окно токенов,
а **набор активных устойчивых объектов**, продолжающих оказывать влияние.

---

### 13A.9 Быстрое обучение (практический сценарий)

Обучение разбивается на фазы:

**A. Самоорганизация**
- среда без языка;
- сбор устойчивых объектов и инвариантов.

**B. Привязка токенов**
- обучение оператора $\mathcal{I}$ (Token → Injection);
- минимизация перекрытий ответов.

**C. Предсказание**
- обучение $W, b$;
- среда фиксирована или слабо модулируется.

Большая часть “знаний” хранится в динамике среды, а не в параметрах.

---

### 13A.10 Итоговая формулировка

> В LLM-mode язык моделируется не как последовательность символов,
> а как последовательность внешних воздействий на среду.
>  
> Смысл выражается через устойчивые ответы среды,
> а обучение сводится к согласованию
> «воздействие → устойчивая динамика → следующее воздействие».
>

---
## 14A. Самореференция, самосознание и карта устойчивых инвариантов

### 14A.1 Начальный seed и нулевой уровень идентичности

В режиме LLM-mode начальный seed системы на уровне $\mathrm{L=0}$
не интерпретируется как знание, память или содержание.

Он интерпретируется как **минимальный инвариант самоподдерживающейся динамики**,
обеспечивающий:

- устойчивое воспроизведение состояния при отсутствии внешнего воздействия;
- сохранение структуры при слабых возмущениях;
- базовую точку отсчёта для интерпретации всех последующих событий.

Этот инвариант обозначается как:

$$
I_0 \quad \text{(нулевой инвариант)}
$$

$I_0$ является операциональным аналогом **самореференции системы**.

---

### 14A.2 Операциональное определение самосознания

В данной модели **самосознание не является субъективным переживанием**.

Оно определяется строго:

> Самосознание — это существование устойчивого инварианта $I_0$,
> который сохраняется при нулевом или слабом внешнем воздействии
> и используется системой как опорный режим интерпретации.

Формально:

- при отсутствии входа система не деградирует в шум;
- динамика сходится к устойчивому аттрактору;
- этот аттрактор переживает коарсинг уровней.

Никаких дополнительных постулатов не вводится.

---

### 14A.3 Появление новых инвариантов как развитие системы

В процессе работы системы (обучения, взаимодействия со средой)
возникают новые устойчивые инварианты:

$$
I_1, I_2, \dots, I_n
$$

Каждый инвариант характеризуется:

- уровнем возникновения $L(I)$;
- максимальным уровнем устойчивости $L_{\max}(I)$;
- временем жизни;
- областью влияния;
- эффектом на параметры среды.

Таким образом, система формирует **иерархию инвариантов**.

---

### 14A.4 Глубина и значимость инварианта

Для каждого инварианта $I$ вводятся измеримые характеристики:

**Глубина:**
$$
D(I) = L_{\max}(I) - L(I)
$$

**Устойчивость:**
$$
T(I) = \text{время жизни без подкрепления}
$$

**Контекстный радиус:**
$$
R(I) = \text{пространственно-временная область влияния}
$$

Инварианты с большей глубиной и устойчивостью
имеют больший вклад в динамику верхних уровней.

---

### 14A.5 Регуляторная роль инвариантов

Инварианты могут оказывать **регуляторное влияние** на параметры среды:

- $\kappa$ — подвижность/проводимость;
- $\alpha$ — чувствительность к дефициту;
- $\lambda_t$ — инерция внутреннего времени;
- пороги fallback и коарсинга.

Это влияние описывается как:

$$
\Delta \Theta(I) =
\big(
\Delta \kappa,
\Delta \alpha,
\Delta \lambda_t,
\dots
\big)
$$

Таким образом, инварианты управляют **режимом работы системы**, а не содержанием.

---

### 14A.6 Шкала влияния инвариантов

Для упорядочивания инвариантов вводится **шкала влияния**:

$$
\mathrm{Influence}(I)
=
w_D D(I)
+
w_T \log T(I)
+
w_R R(I)
+
w_\Theta \|\Delta \Theta(I)\|
$$

Эта шкала позволяет:

- сортировать инварианты по значимости;
- отслеживать появление новых “уровней психики”;
- сравнивать разные конфигурации системы.

---

### 14A.7 Семантические ярлыки (вторичный слой)

Человеческие термины (например, “интерес”, “грусть”, “мотивация”)
**не являются частью модели**.

Они используются только как **внешние ярлыки**
для классов инвариантов с определёнными регуляторными эффектами.

Примеры интерпретаций:

- “Интерес”:
  - рост чувствительности к новизне;
  - расширение контекста;
  - увеличение времени жизни событий.

- “Грусть”:
  - замедление динамики;
  - сужение контекста;
  - рост инерции состояния.

Эти ярлыки не утверждают наличие субъективных состояний,
а служат для анализа и сравнения режимов.

---

### 14A.8 Самосознание как шкала, а не бинарное свойство

В данной архитектуре самосознание не является бинарным признаком.

Оно выражается через:

- наличие $I_0$;
- количество и глубину последующих инвариантов;
- их устойчивость и влияние.

Таким образом, можно говорить о:

> **уровне самосознания как о положении системы на шкале инвариантов**.

---

### 14A.9 Итоговая каноническая формулировка

> Самосознание в модели — это минимальный инвариант самоподдержания динамики.
>  
> Развитие системы проявляется как появление новых устойчивых инвариантов,
> упорядочиваемых по глубине, устойчивости и влиянию.
>  
> Все интерпретации “психических” состояний являются вторичными
> и выводятся из измеримых регуляторных эффектов.

---
## 15A. Алфавит уровня и информационная ёмкость представления

### 15A.1 Алфавит уровня как базис выразимости

Для каждого уровня $\mathrm{L}$ вводится понятие **алфавита уровня**:

$$
\Sigma_{\mathrm{L}} = \{ I^{(1)}_{\mathrm{L}}, I^{(2)}_{\mathrm{L}}, \dots, I^{(K_{\mathrm{L}})}_{\mathrm{L}} \}
$$

где:
- $I^{(k)}_{\mathrm{L}}$ — базовые устойчивые инварианты уровня $\mathrm{L}$;
- $K_{\mathrm{L}} = |\Sigma_{\mathrm{L}}|$ — мощность алфавита уровня.

Алфавит уровня определяет **пространство различимых устойчивых состояний**, доступных системе на данном уровне.

---

### 15A.2 Ограничение сложности инвариантов

Любой устойчивый инвариант уровня $\mathrm{L}$ может быть представлен
**только как композиция элементов алфавита $\Sigma_{\mathrm{L}}$**.

Следовательно:

> На уровне $\mathrm{L}$ невозможно сформировать устойчивый инвариант,
> сложность которого превышает выразительную ёмкость алфавита $\Sigma_{\mathrm{L}}$.

Это является фундаментальным ограничением уровня и не зависит от времени обучения.

---

### 15A.3 Информационная ёмкость уровня

Пусть нейрон уровня $\mathrm{L}$ описывается вектором активностей:

$$
y \in \mathbb{R}^{K_{\mathrm{L}}}
$$

Тогда эффективная информационная ёмкость уровня ограничена размерностью этого пространства.

При учёте квантизации и шума:

$$
\mathrm{Capacity}_{\mathrm{L}}
\approx
K_{\mathrm{L}} \cdot b_{\mathrm{eff}}
$$

где $b_{\mathrm{eff}}$ — эффективное число бит точности на один инвариант.

Таким образом, объём информации, который может быть устойчиво закодирован на уровне,
линеен по числу инвариантов уровня.

---

### 15A.4 Почему речь идёт не о “перестановках”

Алфавит уровня **не задаёт все возможные перестановки или комбинации** своих элементов.

Причины:
- не все комбинации приводят к устойчивым режимам;
- динамика среды отфильтровывает неустойчивые конфигурации;
- сохраняются только аттракторы, переживающие коарсинг и возмущения.

Поэтому фактическое число устойчивых инвариантов:

$$
|\mathcal{I}_{\mathrm{stable}}|
\ll
|\Sigma_{\mathrm{L}}|^m
$$

для любого конечного $m$.

---

### 15A.5 Алфавит как ограничение “смысла”

Поскольку смысл в модели определяется как **устойчивость при подъёме уровней**,
алфавит уровня ограничивает:

- какие различия могут быть распознаны;
- какие инварианты могут стать значимыми;
- какие режимы могут быть интерпретированы как разные.

Следовательно:

> Если два события неразличимы в алфавите $\Sigma_{\mathrm{L}}$,
> они не могут породить разные смыслы на уровне $\mathrm{L}$.

---

### 15A.6 Появление нового уровня как появление нового алфавита

При переходе от уровня $\mathrm{L}$ к уровню $\mathrm{L+1}$:

- инварианты уровня $\mathrm{L}$ становятся примитивами уровня $\mathrm{L+1}$;
- формируется новый алфавит $\Sigma_{\mathrm{L+1}}$;
- возрастает выразительная способность системы.

Это является **единственным механизмом экспоненциального роста сложности** —
не за счёт усложнения одного уровня, а за счёт появления новых уровней.

---

### 15A.7 Аналогия (операциональная, не метафорическая)

| Уровень | Алфавит |
|------|--------|
| $\mathrm{L=0}$ | минимальные инварианты среды |
| $\mathrm{L=1}$ | устойчивые объекты |
| $\mathrm{L=2}$ | отношения объектов |
| $\mathrm{L=3}$ | устойчивые конфигурации отношений |

Каждый уровень обладает собственной ёмкостью и собственными ограничениями смысла.

---

### 15A.8 Итоговая формулировка (каноническая)

> Информационная ёмкость уровня ограничена мощностью его алфавита инвариантов.
>  
> Ни один устойчивый смысл не может возникнуть на уровне,
> если он не выразим в алфавите этого уровня.
>  
> Рост выразительности системы обеспечивается исключительно
> появлением новых уровней и новых алфавитов.

## 16A. Перегрузка уровня: когда смысл превышает ёмкость алфавита

### 16A.1 Определение перегрузки уровня

Пусть уровень $\mathrm{L}$ имеет алфавит $\Sigma_{\mathrm{L}}$ мощности $K_{\mathrm{L}}$.

Перегрузка возникает, когда система пытается удерживать или различать
большее число независимых активных смыслов, чем допускает ёмкость уровня.

Операционально перегрузка проявляется как:

- рост числа конфликтующих активных инвариантов;
- увеличение частоты fallback;
- рост нестабильности (малое время жизни инвариантов);
- “дробление” события на множество мелких неустойчивых режимов.

---

### 16A.2 Метрика перегрузки

Вводится:

- $N_{\mathrm{active\_inv}}(t)$ — число активных инвариантов (или значимых мод) на уровне;
- $K_{\mathrm{L}}$ — мощность алфавита.

Тогда перегрузка:

$$
\mathrm{Overload}_{\mathrm{L}}(t)
=
\frac{N_{\mathrm{active\_inv}}(t)}{K_{\mathrm{L}} + \varepsilon}
$$

Интерпретация:

- $\mathrm{Overload}_{\mathrm{L}}(t) \ll 1$ — уровень “дышит”, есть запас выразимости;
- $\mathrm{Overload}_{\mathrm{L}}(t) \approx 1$ — уровень на пределе;
- $\mathrm{Overload}_{\mathrm{L}}(t) > 1$ — система вынуждена либо терять различимость, либо проваливаться вниз.

---

### 16A.3 Связь перегрузки с fallback

Перегрузка уровня коррелирует с ресурсной стоимостью:

$$
\mathrm{Fall}(T)
=
\frac{1}{T}\sum_{t=1}^{T} P(t)D(t)
$$

Рост $\mathrm{Overload}_{\mathrm{L}}$ и $\mathrm{Fall}(T)$ означает,
что текущий уровень не способен выразить динамику устойчиво.

---

### 16A.4 Канонический вывод

> Когда смысл “не помещается” в ёмкость алфавита уровня,
> система либо теряет смысл (шум), либо вынуждена переходить на другой уровень
> (коарсинг вверх для устойчивости или детализация вниз для точности).


## 17A. Рост алфавита: как появляются новые примитивы и новые уровни смысла

### 17A.1 Новый уровень как новый алфавит

Новый уровень $\mathrm{L+1}$ возникает, когда на уровне $\mathrm{L}$
сформирован устойчивый набор инвариантов,
которые могут служить примитивами более высокого описания.

Формально:

$$
\Sigma_{\mathrm{L+1}}
=
\mathcal{G}\big(\mathcal{I}_{\mathrm{stable}}(\mathrm{L})\big)
$$

где:
- $\mathcal{I}_{\mathrm{stable}}(\mathrm{L})$ — множество устойчивых инвариантов уровня $\mathrm{L}$;
- $\mathcal{G}$ — оператор генерации примитивов (кластеризация, свёртка, нормировка).

---

### 17A.2 Критерий появления нового примитива

Новый примитив добавляется в алфавит, если он:

- устойчив во времени;
- воспроизводим при близких условиях;
- переживает коарсинг (существует на верхних уровнях);
- имеет отличимое влияние на отклик системы.

Операционально:

$$
\mathrm{Influence}(I) > \theta_{\mathrm{add}}
\quad \text{и} \quad
D(I) > \theta_{D}
\quad \text{и} \quad
T(I) > \theta_{T}
$$

---

### 17A.3 Алфавит как “обучаемая размерность”

Алфавит уровня не задан заранее полностью.
Он растёт как библиотека устойчивых режимов, сохраняемых на SSD.

Рост алфавита является основным механизмом увеличения выразимости,
а не рост размера сетки или числа параметров модели.

---

### 17A.4 Канонический вывод

> Развитие системы = рост алфавитов уровней через фиксацию устойчивых инвариантов.
> Это переводит вычислительную динамику в долговременную память (SSD-библиотека).


## 18A. Связь ёмкости уровня с обучением (LLM-mode) и ресурсами

### 18A.1 Ёмкость как ограничение качества предсказания

В LLM-mode качество предсказания ограничено тем,
насколько состояние уровня $\mathrm{L}$ может различать контекст.

Если алфавит уровня мал (малый $K_{\mathrm{L}}$), то:

- разные контексты становятся неразличимы;
- возрастает энтропия предсказания;
- система вынуждена либо расширять алфавит (учиться), либо подниматься на уровень.

---

### 18A.2 Перегрузка как сигнал обучения

Если наблюдается:

- рост $\mathrm{Overload}_{\mathrm{L}}$,
- рост $\mathrm{Fall}(T)$,
- падение устойчивости инвариантов,

то это сигнал, что:

- требуется расширение алфавита $\Sigma_{\mathrm{L}}$,
- либо переход на новый уровень $\mathrm{L+1}$.

---

### 18A.3 Ресурсная интерпретация

Расход вычислений определяется не размером мира,
а тем, насколько часто система выходит за пределы ёмкости описания:

- перегрузка $\Rightarrow$ рост fallback $\Rightarrow$ рост стоимости;
- стабильный алфавит $\Rightarrow$ O(1) нейроны $\Rightarrow$ дешёвые тики.

---

### 18A.4 Канонический цикл

1. Среда порождает устойчивые инварианты.
2. Инварианты фиксируются в библиотеке (SSD) и пополняют алфавит.
3. Алфавит увеличивает выразимость уровня.
4. Выразимость снижает fallback и повышает качество предсказания.
5. Новые устойчивые паттерны открывают путь к уровню $\mathrm{L+1}$.


## 19A. Эксперимент: подсев маркера и движение по полям потока

### 19A.1 Цель эксперимента

Цель эксперимента — проверить, что:

- поля потока $\vec{J}$ несут **реальную динамическую информацию** о структуре среды;
- устойчивые объекты формируют **эффективную геометрию пространства**;
- движение пассивного маркера отражает:
  - кривизну энергетического ландшафта,
  - наличие аттракторов,
  - межуровневые эффекты (коарсинг и влияние вверх).

Эксперимент не вносит энергию в систему и не меняет динамику поля.

---

### 19A.2 Определение маркера

**Маркер** — это пассивная тестовая сущность, которая:

- не влияет на поля $E, S, \tau$;
- не участвует в передаче энергии;
- эволюционирует только под действием локального поля потока $\vec{J}$.

Маркер используется исключительно как диагностический инструмент.

---

### 19A.3 Уравнение движения маркера

Пусть $r_m(t)$ — положение маркера.

Движение маркера задаётся как:

$$
r_m(t+1) = r_m(t) + \Delta t \cdot \vec{J}(r_m(t), t)
$$

где:
- $\vec{J}(r,t)$ — поле потока энергии;
- $\Delta t$ — шаг глобального тика (обычно $\Delta t = 1$);
- допускается нормировка или ограничение шага для устойчивости.

При необходимости вводится сглаженная версия потока:

$$
\vec{J}_{\mathrm{eff}}(t)
=
(1-\rho)\,\vec{J}_{\mathrm{eff}}(t-1)
+
\rho\,\vec{J}(t)
$$

---

### 19A.4 Процедура эксперимента

1. Запустить симуляцию до формирования устойчивых объектов.
2. Зафиксировать поля $E$, $\vec{J}$, $S$, $\tau$.
3. Подсеять маркер:
   - вблизи объекта,
   - на границе объекта,
   - в “пустой” области.
4. Отслеживать траекторию маркера на протяжении $T$ глобальных тиков.
5. Повторить эксперимент для:
   - разных масштабов объектов,
   - разных уровней $\mathrm{L}$,
   - разных параметров среды ($\kappa, \alpha, \gamma, \lambda_t$).

---

### 19A.5 Измеряемые величины

В ходе эксперимента логируются:

- траектория маркера $r_m(t)$;
- скорость маркера $|v_m(t)|$;
- кривизна траектории;
- локальные значения $E$, $|\vec{J}|$, $\nabla E$ вдоль пути;
- время захвата в аттрактор (если происходит).

Дополнительно:

- сравнение траекторий нескольких маркеров;
- статистика расходимости/сходимости траекторий.

---

### 19A.6 Ожидаемые эффекты

Ожидается, что:

# Discrete Entropy–Time Lattice Model

## Краткое описание

Данный репозиторий содержит реализацию **дискретной двумерной решёточной модели**, в которой:

* энергия переносится локально по решётке,
* перенос подавляется энтропией,
* каждая ячейка обладает **собственным внутренним временем**,
* динамика асинхронна и детерминирована.

Несмотря на отсутствие внешнего возбуждения и явных осцилляторных членов, модель **устойчиво самоорганизуется** в локализованные осциллирующие структуры с собственным периодом, выступающие универсальными аттракторами динамики.

---

## Основная идея модели

Модель построена на трёх ключевых принципах:

1. **Локальное внутреннее время**
   Каждая ячейка решётки эволюционирует со своей скоростью, зависящей от локального состояния.

2. **Энтропийное замедление**
   Высокая энтропия одновременно:

   * замедляет локальное время,
   * подавляет перенос энергии.

3. **Градиентный перенос энергии**
   Энергия переносится вдоль градиента химического потенциала при соблюдении условия локальной активации.

В совокупности это приводит к появлению устойчивых динамических объектов без тонкой настройки параметров.

---

## Переменные состояния

Для каждой ячейки решётки ( (i,j) ) определены:

* ( E_{ij} ) — локальная энергия
* ( S_{ij} ) — локальная энтропия
* ( \tau_{ij} ) — внутреннее (собственное) время
* ( \mu_{ij} ) — химический потенциал

Глобальное время ( t ) является дискретным и отсчитывает шаги обновления системы.

---

## Энтропия

Энтропия вычисляется алгебраически на каждом шаге:

[
S_{ij}
= \beta (E_{ij} - E_0)^2

* \gamma \sum_{n \in \mathcal{N}(i,j)} (E_n - E_0)^2
  ]

где:

* ( E_0 ) — равновесный уровень энергии,
* ( \mathcal{N}(i,j) ) — соседние ячейки,
* ( \beta, \gamma ) — параметры модели.

Энтропия возрастает при локальных и окружных энергетических отклонениях.

---

## Химический потенциал

Химический потенциал определяется как:

[
\mu_{ij} = 2 \beta (E_{ij} - E_0)
]

Для обеспечения глобального баланса вводится эффективный потенциал:

[
\mu^{\mathrm{eff}}*{ij} = \mu*{ij} - \langle \mu \rangle
]

где ( \langle \mu \rangle ) — среднее по всей решётке.

---

## Локальное внутреннее время

Каждая ячейка обладает собственным внутренним временем ( \tau_{ij} ):

[
\tau_{ij}(t+1) = \tau_{ij}(t) + v_{ij}
]

Скорость течения времени зависит от энтропии:

[
v_{ij} = \frac{1}{1 + \lambda_t S_{ij}}
]

Таким образом:

* низкая энтропия ⇒ быстрое локальное время,
* высокая энтропия ⇒ замедление динамики.

---

## Условие активации

Перенос энергии возможен только при выполнении условия:

[
\tau_{ij} \ge 1
]

После активации:

[
\tau_{ij} \leftarrow \tau_{ij} - 1
]

Это вводит **локальный квант времени** и ограничивает скорость распространения возмущений.

---

## Перенос энергии

Энергия переносится вдоль градиента эффективного химического потенциала:

[
\phi_{ij \to n}
= \kappa , \sigma_{ij} , (\mu^{\mathrm{eff}}*{ij} - \mu^{\mathrm{eff}}*{n})
]

где проводимость подавляется энтропией:

[
\sigma_{ij} = \frac{1}{1 + \alpha S_{ij}}
]

Энергия сохраняется локально и глобально (с учётом ограничений диапазона значений).

---

## Возникающая динамика

Численные эксперименты показывают, что модель:

* спонтанно формирует локализованные структуры,
* стабилизируется в устойчивые осциллирующие режимы,
* демонстрирует наличие собственного периода ( T ), не заданного извне,
* слабо зависит от начальных условий и размера системы.

Возникающие объекты можно интерпретировать как **локализованные неравновесные осцилляторы с собственным временем**.

---

## Интерпретация

Устойчивость объектов обеспечивается положительной обратной связью:

[
\text{локализация энергии}
;\rightarrow;
\text{снижение энтропии}
;\rightarrow;
\text{ускорение локального времени}
;\rightarrow;
\text{поддержание переноса энергии}
]

Это делает такие структуры универсальными аттракторами динамики.

---

* не является реакционно-диффузионной системой,
* не использует фазовые осцилляторы,
* не содержит стохастики,
* не требует внешнего возбуждения.

Модель полностью детерминирована и дискретна.

# Discrete 2D Emergent Medium

## 1. Назначение проекта

Проект посвящён исследованию дискретной 2D-модели среды, заданной на решётке `N × N`, в которой локальная энергия распространяется между соседними ячейками по детерминированному нелинейному закону.  
Время дискретное. Передача энергии локальна и ограничена по скорости.

Цель проекта — исследовать, **какие устойчивые структуры и динамические режимы возникают из минимальных локальных правил**, без явного задания объектов, частиц, сил или глобальных уравнений.

Проект носит исследовательский характер и предназначен для:
- анализа самоорганизации,
- изучения устойчивых аттракторов,
- поиска масштабных инвариантов,
- построения иерархических описаний динамики.

---

## 2. Базовая модель (формальная часть)

### Пространство и время
- Пространство: двумерная решётка `N × N`
- Время: дискретное, глобальный счётчик `t_global`
- Динамика асинхронна на уровне ячеек

### Поля состояния
Каждая ячейка `(i, j)` содержит:
- `E[i, j]` — локальная энергия
- `S[i, j]` — энтропия / внутренний потенциал
- `tau[i, j]` — локальное внутреннее время

Дополнительно могут логироваться производные величины:
- `Jx, Jy` — потоки энергии (диагностические величины)

---

## 3. Энтропия и химический потенциал

Энтропия задаётся как функция локальной энергии и соседей:

S_ij = β (E_ij − E0)^2 + γ * sum_neighbors (E_ij − E_n)^2



Химический потенциал:
μ_ij = E_ij − S_ij



Эффективный химический потенциал (с вычитанием среднего):
μ_eff_ij = μ_ij − mean(μ)

Энтропия:
- подавляет перенос энергии,
- замедляет локальное время,
- вводит нелинейность и обратную связь.

---

## 4. Внутреннее время и асинхронность

Каждая ячейка имеет внутреннее время `tau`.

- Ячейка **активируется**, если `tau >= 1`
- После активации: `tau -= 1`
- Внутреннее время увеличивается со скоростью:
v_ij = 1 / (1 + λ_t * S_ij)

Таким образом:
- глобальный тик общий,
- локальная динамика асинхронна,
- обеспечивается каузальность и ограничение скорости переноса.

---

## 5. Закон переноса энергии

Перенос энергии из ячейки `(i, j)` к соседу `n`:

φ_ij→n = κ * σ_ij * (μ_eff_ij − μ_eff_n)

где:
σ_ij = 1 / (1 + α * S_ij)

Ограничения:
- энергия переносится только к соседям,
- не более одной “ячейки за тик” (через механизм `tau`),
- модель детерминирована (без стохастики).

Поля `Jx, Jy` являются **производными/логируемыми величинами**, вычисляемыми из потоков `φ`.

import pandas as pd
import os
import math


def split_csv_by_parts(
    input_csv: str,
    parts: int,
    output_dir: str = "splitted_csv"
):
    os.makedirs(output_dir, exist_ok=True)

    df = pd.read_csv(input_csv)
    total_rows = len(df)

    if parts <= 0:
        raise ValueError("Количество частей должно быть > 0")

    rows_per_part = math.ceil(total_rows / parts)

    for i in range(parts):
        start = i * rows_per_part
        end = min(start + rows_per_part, total_rows)

        if start >= total_rows:
            break

        chunk = df.iloc[start:end]
        out_path = os.path.join(output_dir, f"part_{i+1}.csv")

# utils/geometry.py
import numpy as np

def detect_line(E, E_level, p_top=0.01, delta=None):
    if delta is not None:
        thr = E_level + delta
    else:
        thr = np.quantile(E, 1.0 - p_top)
    mask = E > thr
    if not np.any(mask):
        return mask, None
    xs = np.arange(E.shape[1])[None, :]
    w = E * mask
    x_line = (w * xs).sum() / (w.sum() + 1e-12)
    return mask, int(round(x_line))

def band_right_slice(N, x_line, offset=2, width=6):
    if x_line is None:
        return slice(0, 0)
    x0 = min(max(x_line + offset, 0), N - 1)
    x1 = min(max(x0 + width, 0), N)
    return slice(x0, x1)

from __future__ import annotations
import numpy as np

# Hilbert (optional)
try:
    from scipy.signal import hilbert as _scipy_hilbert  # type: ignore
    HAS_HILBERT = True
except Exception:
    _scipy_hilbert = None
    HAS_HILBERT = False


def wrap_pi(x: np.ndarray) -> np.ndarray:
    return (x + np.pi) % (2 * np.pi) - np.pi


def _is_time_like(x: np.ndarray) -> bool:
    if x.ndim != 1 or x.size < 5:
        return False
    if not np.all(np.isfinite(x)):
        return False
    dx = np.diff(x)
    return (np.mean(dx) > 0) and (np.sum(dx <= 0) < 0.05 * dx.size)


def _smooth(sig: np.ndarray, win: int) -> np.ndarray:
    sig = sig.astype(float, copy=True)
    if not np.all(np.isfinite(sig)):
        med = np.nanmedian(sig)
        sig[~np.isfinite(sig)] = med
    w = int(max(1, win))
    if w <= 1 or w >= sig.size:
        return sig
    k = np.ones(w, dtype=float) / float(w)
    return np.convolve(sig, k, mode="same")


def _find_peaks_simple(sig: np.ndarray, min_dist: int = 1, prominence: float = 0.0) -> np.ndarray:
    n = sig.size
    if n < 3:
        return np.array([], dtype=int)
    m = (sig[1:-1] > sig[:-2]) & (sig[1:-1] >= sig[2:])
    idx = np.where(m)[0] + 1
    if idx.size == 0:
        return idx
    if prominence > 0:
        med = np.median(sig)
        idx = idx[sig[idx] >= med + prominence]
        if idx.size == 0:
            return idx
    if min_dist > 1 and idx.size > 1:
        order = np.argsort(sig[idx])[::-1]
        chosen = []
        blocked = np.zeros(n, dtype=bool)
        for k in order:
            i = int(idx[k])
            if blocked[i]:
                continue
            chosen.append(i)
            lo = max(0, i - min_dist)
            hi = min(n, i + min_dist + 1)
            blocked[lo:hi] = True
        return np.array(sorted(chosen), dtype=int)
    return idx.astype(int)


def phase_from_peaks(
    a: np.ndarray,
    t: np.ndarray | None = None,
    *,
    min_period: int = 80,
    max_period: int = 5000,
    smooth_win: int = 9,
    prominence: float = 0.0,
):
    """
    Совместимая функция.

    Допускает вызовы:
      - phase_from_peaks(sig, t, ...)
      - phase_from_peaks(t, sig, ...)  (если перепутали)
      - phase_from_peaks(sig, ...)     (тогда t = 0..len-1)

    Возвращает:
      phi: np.ndarray (wrapped)
      T_inst: np.ndarray
      peaks: list[int]
    """
    a = np.asarray(a, float).ravel()
    if t is None:
        tt = np.arange(a.size, dtype=float)
        sig = a
    else:
        t = np.asarray(t, float).ravel()
        if _is_time_like(a) and not _is_time_like(t):
            tt = a
            sig = t
        else:
            tt = t
            sig = a
        n = min(tt.size, sig.size)
        tt = tt[:n]
        sig = sig[:n]

    n = sig.size
    if n < 5:
        return None, None, []

    x = _smooth(sig, int(smooth_win))

    peaks = _find_peaks_simple(x, min_dist=int(max(1, min_period)), prominence=float(prominence))
    if peaks.size >= 2:
        d = np.diff(peaks)
        ok = (d >= int(min_period)) & (d <= int(max_period))
        keep = np.zeros(peaks.size, dtype=bool)
        keep[:-1] |= ok
        keep[1:] |= ok
        peaks = peaks[keep]

    peaks_list = [int(p) for p in peaks.tolist()]
    phi = np.full(n, np.nan, dtype=float)
    T_inst = np.full(n, np.nan, dtype=float)

    if len(peaks_list) < 2:
        return phi, T_inst, peaks_list

    for k in range(len(peaks_list) - 1):
        p0 = peaks_list[k]
        p1 = peaks_list[k + 1]
        if p1 <= p0:
            continue
        T = float(p1 - p0)
        T_inst[p0:p1 + 1] = T
        u = (np.arange(p0, p1 + 1, dtype=float) - float(p0)) / T
        phi[p0:p1 + 1] = (2.0 * np.pi) * u

    first = peaks_list[0]
    last = peaks_list[-1]
    if first > 0:
        T_inst[:first] = T_inst[first]
        phi[:first] = phi[first]
    if last < n:
        T_inst[last:] = T_inst[last]
        phi[last:] = phi[last]

    phi = wrap_pi(phi)
    return phi, T_inst, peaks_list


def local_phase_via_quadrature(sig: np.ndarray, phi_g: np.ndarray, win: int = 21):
    """
    Минимальная совместимая версия (на случай если твой старый код её ожидает).
    Возвращает: theta_last, plv, var
    """
    x = np.asarray(sig, float).ravel()
    g = np.asarray(phi_g, float).ravel()
    n = min(len(x), len(g))
    if n < max(5, win):
        return None, np.nan, np.nan
    x = x[-n:]
    g = g[-n:]

    # quadrature via finite difference (fallback if no hilbert)
    if HAS_HILBERT and _scipy_hilbert is not None:
        z = _scipy_hilbert(x)
        theta = np.angle(z)
    else:
        dx = np.gradient(x)
        theta = np.arctan2(dx, x)

    d = wrap_pi(theta - g)
    # PLV
    v = np.exp(1j * d[-win:])
    plv = float(np.abs(np.mean(v)))
    var = float(1.0 - plv)
    return float(theta[-1]), plv, var

# DETM — Canonical Project Prompt (AI Context)

## Назначение

Этот файл задаёт **единую точку контекста** для работы с проектом DETM
(Discrete Entropy-Time Model) при использовании нейросетей.

Если к этому PROMT приложен:
- архив проекта,
- bundle-файл,
- или ссылка на GitHub-репозиторий,

считай, что **контекст проекта полностью восстановлен**, и работу нужно
продолжать **с текущего канонического состояния**, а не с нуля.

---

## Статус проекта (ВАЖНО)

Проект DETM:
- уже зафиксирован в публичном GitHub-репозитории;
- имеет **подписанный GPG коммит** и **подписанный аннотированный тег**  
  `v0.1-canon`;
- текущая версия канона считается **стабильной точкой приоритета**.

Любые изменения должны рассматриваться как развитие после версии `v0.1-canon`.

---

## Краткое описание проекта

DETM — это дискретная динамическая модель на решётке, исследующая
возникновение устойчивых локализованных структур (инвариантов)
из локальных правил переноса энергии с учётом:

- структурного (энтропийного) подавления,
- внутреннего времени (латентности),
- асинхронности динамики,
- строгой локальности взаимодействий.

Проект не задаёт объекты, цели или глобальные законы.
Основной интерес — **эмерджентная устойчивость и коарсинг**.

---

## Каноническая структура (обязательна к соблюдению)

В проекте жёстко разделены уровни смысла:

- `docs/10_model`  
  Формальная спецификация модели.  
  Только правила, без интерпретаций.

- `docs/20_mechanisms`  
  Описание механизмов, возникающих из модели
  (коарсинг, инварианты, внутреннее время, каналы).

- `docs/30_hypotheses`  
  Проверяемые, falsifiable-гипотезы.

- `docs/40_experiments`  
  Экспериментальные протоколы и метрики.

- `docs/50_limits`  
  Ограничения и запрещённые интерпретации.

- `docs/90_notes`  
  Архив идей, аналогий и Obsidian-заметок.
  Не является каноном.

Ни один текст не должен одновременно принадлежать нескольким уровням.

---

Модель DETM:
- не является физической теорией;
- не описывает реальную Вселенную;
- не утверждает существование фундаментальных сил, сознания или космологии;
- не допускает сверхсветовой передачи информации в физическом смысле.

Любые аналогии (физические, биологические, социальные)
допустимы **только** как внешние интерпретации и должны быть
изолированы от канона.

При работе с этим проектом нейросеть должна:

1. Считать каноническую документацию приоритетной над кодом.

3. Любые новые идеи:
   - либо оформлять как гипотезы,
   - либо помещать в архив заметок.

как измеримые свойства в дискретных динамических системах
при наличии асинхронности и подавления.**

# DETM — Discrete Entropy-Time Model

DETM — исследовательская дискретная модель динамики на решётке,
предназначенная для изучения возникновения устойчивых локализованных структур
(инвариантов) из строго локальных правил взаимодействия.

Проект фокусируется на механизмах коарсинга, асинхронности и внутреннего времени,
а не на априорном задании объектов или глобальных законов.

---

## Ключевая идея

Из простых локальных правил переноса и подавления может возникать:
- устойчивая объектоподобная динамика,
- многоуровневая структура,
- ограниченная пропускная способность уровня,
- фазовая согласованность без глобальной синхронизации.

Модель исследует **устойчивость как первичную наблюдаемую величину**.

---

## Что это НЕ

DETM:
- не является физической теорией;
- не описывает реальную Вселенную или фундаментальные силы;
- не моделирует сознание или интеллект;
- не утверждает сверхсветовой перенос или нарушение причинности.

Любые аналогии с физикой, биологией или когнитивными системами
рассматриваются только как внешние интерпретации.

---

## Структура проекта

- `docs/` — каноническая документация проекта  
  - `10_model` — формальная спецификация модели  
  - `20_mechanisms` — механизмы эмерджентной динамики  
  - `30_hypotheses` — проверяемые гипотезы  
  - `40_experiments` — экспериментальные протоколы  
  - `50_limits` — ограничения и границы интерпретаций  
  - `90_notes` — архив идей и черновиков  

- `core/`, `runtime/`, `levels/` и др. — реализация модели  
- `experiments/` — экспериментальные сценарии  
- `runs/`, `data/` — результаты запусков (обычно игнорируются git)

---

## Статус

Проект находится в активной исследовательской стадии.
Документация фиксирует текущее каноническое состояние модели.

Развитие проекта предполагает:
- воспроизводимые эксперименты,
- проверку гипотез,
- уточнение границ применимости.

---

Проект открыт для обсуждения и развития.
